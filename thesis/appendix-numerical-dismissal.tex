\begin{lemma}[Dismissal via constraint-satisfying direction]%
\label{lem:constraint-direction-dismissal}
Let $(S, \sigma)$ be a pair for which the linear
system~\eqref{eq:linear-system} has a solution
$(\beta_0, \lambda_0, \nu_0)$ with $\beta_{0,i} > 0$ for all
$i \in S$.
Let $\delta\beta \in \mathbb{R}^{|S|} \setminus \{0\}$ satisfy
the constraint equations
\[
  N^\top \delta\beta = 0, \qquad \eta^\top \delta\beta = 0.
\]
Define the walk-to-boundary quantities:
for each~$i$ with $\delta\beta_i < 0$, set
$\alpha_i = \beta_{0,i} / |\delta\beta_i|$; then
$\alpha^* = \min\{\alpha_i : \delta\beta_i < 0\}$,
$k$ is an index achieving this minimum,
$\beta^* = \beta_0 + \alpha^*\,\delta\beta$,
and $S' = S \setminus \{k\}$.

Then:
\begin{enumerate}
\item $\delta\beta$ has mixed signs
  (at least one positive and one negative component).
\item $\beta^*_k = 0$, $\beta^*_j \geq 0$ for all $j \in S$,
  and $\beta^*|_{S'}$ is exactly feasible:
  $N^\top \beta^* = 0$ and $\eta^\top \beta^* = 1$
  (which, since $\beta^*_k = 0$, are equivalent to
  the constraints for~$(S', \sigma|_{S'})$).
\item The action value satisfies
  \begin{equation}\label{eq:walk-q-formula}
    \beta^{*\top} H\, \beta^*
    = \nu_0 + (\alpha^*)^2\,
      \delta\beta^\top H\, \delta\beta.
  \end{equation}
\end{enumerate}
In particular, the $(S', \sigma|_{S'})$ algorithm finds value
at least $\nu_0 - (\alpha^*)^2\,
|\delta\beta^\top H\, \delta\beta|$.
\end{lemma}

\begin{proof}
\emph{Part~(1).}
Since $\eta_i = h_{\sigma(i)} > 0$ for all~$i$,
$\sum \eta_i\,\delta\beta_i = 0$ with $\delta\beta \neq 0$
forces at least one positive and one negative component.

\emph{Part~(2).}
The walk-to-boundary construction and non-negativity argument
are identical to Steps~3--4 of
Lemma~\ref{lem:rank-deficiency-dismissal}
(they use only $\beta_0 > 0$, $\delta\beta$ having mixed signs,
and the definition of~$\alpha^*$).
Exact feasibility follows from the constraint equations:
\[
  N^\top\beta^*
  = N^\top\beta_0 + \alpha^*\,N^\top\delta\beta
  = 0 + 0 = 0, \qquad
  \eta^\top\beta^*
  = \eta^\top\beta_0 + \alpha^*\,\eta^\top\delta\beta
  = 1 + 0 = 1.
\]
Since $\beta^*_k = 0$, these equalities pass to the
restriction~$\beta^*|_{S'}$.

\emph{Part~(3).}
Expanding the quadratic form:
\[
  \beta^{*\top} H\,\beta^*
  = \beta_0^\top H\,\beta_0
    + 2\alpha^*\,\delta\beta^\top H\,\beta_0
    + (\alpha^*)^2\,\delta\beta^\top H\,\delta\beta.
\]
The first term equals~$\nu_0$.
For the cross term, the KKT condition
$H\,\beta_0 = N\,\lambda_0 + \eta\,\nu_0$ gives
\[
  \delta\beta^\top H\,\beta_0
  = (N^\top\delta\beta)^\top\lambda_0
    + (\eta^\top\delta\beta)\,\nu_0
  = 0 + 0 = 0. \qedhere
\]
\end{proof}

\begin{lemma}[Constraint projection of approximate null directions]%
\label{lem:constraint-projection}
Let $(\delta\beta, \delta\lambda, \delta\nu)$ be a unit vector
($\|(\delta\beta, \delta\lambda, \delta\nu)\|_2 = 1$) with
residual
\[
  \begin{pmatrix}
    N^\top & 0 & 0 \\
    H & -N & -\eta \\
    \eta^\top & 0 & 0
  \end{pmatrix}
  \begin{pmatrix}\delta\beta\\\delta\lambda\\\delta\nu\end{pmatrix}
  = \begin{pmatrix}r_\beta\\r_\lambda\\r_\nu\end{pmatrix},
  \qquad
  \varepsilon
  := \bigl\|(r_\beta, r_\lambda, r_\nu)\bigr\|_2.
\]
Let $C = [N \mid \eta] \in \mathbb{R}^{|S| \times 5}$
and suppose $C$ has full column rank with smallest singular
value $\sigma_C > 0$.
Define the corrected direction
\[
  \delta\beta'
  = \delta\beta
    - C\,(C^\top C)^{-1}
      \begin{pmatrix}r_\beta \\ r_\nu\end{pmatrix}.
\]
Then:
\begin{enumerate}
\item $N^\top\delta\beta' = 0$ and
  $\eta^\top\delta\beta' = 0$.
\item $\|\delta\beta - \delta\beta'\|_2
  \leq \varepsilon / \sigma_C$.
\item $\delta\beta' \neq 0$ whenever
  $\|\delta\beta\|_2 > \varepsilon / \sigma_C$.
\end{enumerate}
\end{lemma}

\begin{proof}
Write $\Delta = C\,(C^\top C)^{-1}(r_\beta, r_\nu)^\top$.
Since $N^\top\delta\beta = r_\beta$ and
$\eta^\top\delta\beta = r_\nu$, and
$C^\top\Delta = (r_\beta, r_\nu)^\top$ by construction:
\[
  C^\top\delta\beta'
  = C^\top\delta\beta - (r_\beta, r_\nu)^\top
  = (r_\beta, r_\nu)^\top - (r_\beta, r_\nu)^\top = 0,
\]
giving~(1).
For~(2):
$\|\Delta\|
= \|C\,(C^\top C)^{-1}(r_\beta, r_\nu)^\top\|
\leq \sigma_C^{-1}\,\|(r_\beta, r_\nu)\|
\leq \varepsilon / \sigma_C$.
Part~(3) follows by the triangle inequality.
\end{proof}

\begin{proposition}[Approximate dismissal of near-singular systems]%
\label{prop:approximate-dismissal}
Let $(S, \sigma)$ be a pair for which the linear
system~\eqref{eq:linear-system} has a solution
$(\beta_0, \lambda_0, \nu_0)$ with $\beta_{0,i} > 0$ for all
$i \in S$.
Let $(\delta\beta, \delta\lambda, \delta\nu)$ be a unit vector
with residual~$\varepsilon$ as in
Lemma~\ref{lem:constraint-projection},
and suppose $\|\delta\beta\|_2 > \varepsilon / \sigma_C$
(so that the corrected direction~$\delta\beta'$ from
Lemma~\ref{lem:constraint-projection} is non-zero).

Apply Lemma~\ref{lem:constraint-direction-dismissal}
with~$\delta\beta'$: the walk-to-boundary gives
$\beta^*|_{S'} \in M(K)|_{S'}$ with $|S'| = |S| - 1$,
and the $(S', \sigma|_{S'})$ algorithm finds value at least
\[
  \nu_0
    - (\alpha^*)^2\,\|\delta\beta'\|\,
      \Bigl(1 + \frac{\|H\|}{\sigma_C}\Bigr)\,\varepsilon,
\]
where $\|\cdot\|$ denotes the spectral norm.
\end{proposition}

\begin{proof}
Lemma~\ref{lem:constraint-projection} gives $\delta\beta'$
with $N^\top\delta\beta' = 0$,
$\eta^\top\delta\beta' = 0$, and $\delta\beta' \neq 0$.
Lemma~\ref{lem:constraint-direction-dismissal} then gives a
feasible boundary point $\beta^*|_{S'}$ with
\[
  \beta^{*\top} H\,\beta^*
  = \nu_0
    + (\alpha^*)^2\,\delta\beta'^\top H\,\delta\beta'.
\]

It remains to bound
$|\delta\beta'^\top H\,\delta\beta'|$.

Write $\Delta = \delta\beta - \delta\beta'$
with $\|\Delta\| \leq \varepsilon/\sigma_C$
(Lemma~\ref{lem:constraint-projection}).
The second block row of the residual equation gives
$H\,\delta\beta = N\,\delta\lambda + \eta\,\delta\nu + r_\lambda$,
so
\[
  H\,\delta\beta'
  = H\,\delta\beta - H\,\Delta
  = N\,\delta\lambda + \eta\,\delta\nu + r_\lambda - H\,\Delta.
\]
Since $N^\top\delta\beta' = 0$ and $\eta^\top\delta\beta' = 0$:
\begin{align*}
  \delta\beta'^\top H\,\delta\beta'
  &= \underbrace{(N^\top\delta\beta')^\top}_{=\,0}\delta\lambda
    + \underbrace{(\eta^\top\delta\beta')}_{=\,0}\,\delta\nu
    + \delta\beta'^\top(r_\lambda - H\,\Delta) \\
  &= \delta\beta'^\top(r_\lambda - H\,\Delta).
\end{align*}
Therefore
\[
  |\delta\beta'^\top H\,\delta\beta'|
  \leq \|\delta\beta'\|\,
       \bigl(\|r_\lambda\| + \|H\|\,\|\Delta\|\bigr)
  \leq \|\delta\beta'\|\,
       \Bigl(1 + \frac{\|H\|}{\sigma_C}\Bigr)\,\varepsilon,
\]
using $\|r_\lambda\| \leq \varepsilon$ and
$\|\Delta\| \leq \varepsilon/\sigma_C$.

\emph{Conclusion.}
Since the $(S', \sigma|_{S'})$ algorithm maximises over
all of $M(K)|_{S'}$, it finds value at least
\[
  \beta^{*\top} H\,\beta^*
  \geq \nu_0
    - (\alpha^*)^2\,\|\delta\beta'\|\,
      \Bigl(1 + \frac{\|H\|}{\sigma_C}\Bigr)\,\varepsilon.
\]
\end{proof}

\begin{algorithm}[Near-singular system handling]%
\label{alg:near-singular-handling}
For each pair $(S, \sigma)$, the numerical algorithm solves
the KKT system~\eqref{eq:linear-system} via SVD.

\smallskip\noindent
\emph{Setup.}
Let $C = [N \mid \eta] \in \mathbb{R}^{|S| \times 5}$
be the constraint matrix, with smallest singular
value~$\sigma_C$.
Let $\sigma_1 \geq \cdots \geq \sigma_{m+5}$ be the singular
values of the KKT system matrix.
A singular value~$\sigma_j$ is \emph{small} if
$\sigma_j < \sigma_1 \cdot \tau$ for a tolerance~$\tau$;
the corresponding right singular vectors
$v_j = (\delta\beta^{(j)}, \delta\lambda^{(j)},
\delta\nu^{(j)})$ span the \emph{near-null space}.
We say that a near-null direction~$v_j$ has
\emph{significant $\beta$-component} if
$\|\delta\beta^{(j)}\| > \sigma_j / \sigma_C$.

\smallskip\noindent
\emph{Normal.}
If no singular value is small, the system is
well-conditioned and the algorithm uses $\beta_0$ directly.

\smallskip\noindent
\emph{Dismiss.}
If $C$ has full column rank ($\sigma_C > 0$) and some
near-null direction~$v_j$ has significant
$\beta$-component,
the system is approximately rank-deficient in~$\beta$.
Proposition~\ref{prop:approximate-dismissal} applies:
the value is covered (up to a bounded error) by a pair
$(S', \sigma|_{S'})$ with $|S'| = |S| - 1$.
The algorithm records the error bound
(Lemma~\ref{lem:dismissal-error-bound}) and skips to the
next pair.

\smallskip\noindent
\emph{Otherwise}, the algorithm uses $\beta_0$ directly.
This covers two sub-cases:
(a)~$C$ is rank-deficient, so the dismissal analysis
does not apply (the constraint projection of
Lemma~\ref{lem:constraint-projection} requires
$\sigma_C > 0$);
(b)~every near-null direction has
$\|\delta\beta^{(j)}\| \leq \sigma_j / \sigma_C$,
meaning the near-null space is confined to the
multiplier block~$(\lambda, \nu)$ and $\beta_0$ is
well-determined despite the near-singularity.

\smallskip\noindent
\emph{Implementation note.}
The Rust code detects small singular values via a gap
ratio $\sigma_{j-1}/\sigma_j > \tau_{\mathrm{gap}}$,
which is a sufficient but not necessary condition.
A condition-number threshold
$\sigma_j < \sigma_1 \cdot \tau$ is more principled.
The mathematical argument
(Lemma~\ref{lem:dismissal-error-bound}) is valid whenever
the dismiss condition holds, regardless of the detection
method.
\end{algorithm}

\begin{lemma}[Error bound for near-singular dismissal]%
\label{lem:dismissal-error-bound}
Suppose Algorithm~\ref{alg:near-singular-handling}
dismisses a pair $(S, \sigma)$ via a near-null direction
$v_j = (\delta\beta^{(j)}, \delta\lambda^{(j)},
\delta\nu^{(j)})$ with singular value~$\sigma_j$.
Let $(\beta_0, \lambda_0, \nu_0)$ be the computed solution
with $\beta_{0,i} > 0$ for all~$i$, and let
$\delta\beta'$ be the corrected direction from
Lemma~\ref{lem:constraint-projection} applied to~$v_j$.
Then there exists a pair $(S', \sigma|_{S'})$ with
$|S'| = |S| - 1$ whose algorithm finds value at least
\begin{equation}\label{eq:dismissal-error-bound}
  \nu_0
    - (\alpha^*)^2\,\|\delta\beta'\|\,
      \Bigl(1 + \frac{\|H\|}{\sigma_C}\Bigr)\,\sigma_j,
\end{equation}
where $\alpha^* = \min\{\beta_{0,i}/|\delta\beta'_i|
: \delta\beta'_i < 0\}$.
In particular, the maximum over all non-dismissed pairs
is at least~\eqref{eq:dismissal-error-bound}.
\end{lemma}

\begin{proof}
The right singular vector for~$\sigma_j$ satisfies
the residual equation with $\varepsilon = \sigma_j$
(by definition of the SVD).
Since $\|\delta\beta^{(j)}\| > \sigma_j / \sigma_C$,
Lemma~\ref{lem:constraint-projection}(3) gives
$\delta\beta' \neq 0$.
The bound then follows from
Proposition~\ref{prop:approximate-dismissal} with
$\varepsilon = \sigma_j$.
\end{proof}

\begin{remark}[Practical aspects of near-singular handling]%
\label{rem:approximate-dismissal-algorithm}
The bound~\eqref{eq:dismissal-error-bound} can be evaluated
numerically for each dismissed system to confirm the value
loss is negligible.
In the ``otherwise'' case of
Algorithm~\ref{alg:near-singular-handling} (near-null space
confined to multipliers),
the multipliers $(\lambda, \nu)$ are ambiguous but $\beta_0$
is well-determined, so the computed
$\beta_0^\top H\,\beta_0$ is reliable despite the
near-singularity.
The three-valued predicates
(Section~\ref{app:three-valued}) then classify the
candidate based on $\beta_0$ alone, with no additional
uncertainty from the near-singular system.
\end{remark}
