\begin{unverified}
\begin{lemma}[Containment after removing near-redundant facets]
\label{lem:hrep-facet-removal-containment}
Let $a_1,\ldots,a_F \in \R^4$ and
\[
  P = \{x \in \R^4 : \langle a_i,x\rangle \leq 1
      \text{ for } i=1,\ldots,F\}.
\]
Let $R \subseteq \{1,\ldots,F\}$ be the retained indices, let
$S=\{1,\ldots,F\}\setminus R$, and let
\[
  P_R = \{x \in \R^4 : \langle a_i,x\rangle \leq 1
      \text{ for } i\in R\}.
\]
Assume $\delta \geq 0$ and
\[
  \langle a_j,x\rangle \leq 1+\delta
  \quad\text{for every } j\in S \text{ and every } x\in P_R .
\]
Then
\[
  P \subseteq P_R \subseteq (1+\delta)P .
\]
\end{lemma}
\end{unverified}

\begin{proof}
The first inclusion follows because $P$ satisfies every retained inequality.
For the second inclusion, take $x\in P_R$ and put
$y=x/(1+\delta)$. For $i\in R$,
\[
  \langle a_i,y\rangle
  = \frac{\langle a_i,x\rangle}{1+\delta}
  \leq \frac{1}{1+\delta}
  \leq 1.
\]
For $j\in S$, the hypothesis gives
\[
  \langle a_j,y\rangle
  = \frac{\langle a_j,x\rangle}{1+\delta}
  \leq 1.
\]
Thus $y\in P$, so $x=(1+\delta)y\in (1+\delta)P$.
\end{proof}

\begin{unverified}
\begin{fact}[Minkowski--Radon asymmetry bound]
\label{fact:minkowski-asymmetry-center}
% This is the standard bound s(K) <= dim(K) for the Minkowski measure of
% asymmetry, written in the centered pointwise form used below.
Let $K$ be an $m$-dimensional compact convex body in its affine hull, with
$m\geq 1$. Then there exists a point $z\in K$ such that for every $y\in K$,
\[
  z - \frac{1}{m}(y-z) \in K .
\]
Equivalently, the Minkowski asymmetry of $K$ is at most~$m$; see
\cite[definition of Minkowski asymmetry and the following standard bound]%
{BrandenbergGonzalezMerino2017Asymmetry}.
\end{fact}
\end{unverified}

\begin{unverified}
\begin{definition}[Single-band and multi-band guards]
\label{def:single-multi-band-guards}
Let $K$ be a compact convex body. Let
$f_1,\ldots,f_N:\aff(K)\to\R$ be affine functions satisfying
$f_j(y)\leq 1$ for every $y\in K$ and every~$j$. For $r\in\R$, define:
\[
\begin{aligned}
  \operatorname{Single}_K(r)
  &\iff
  \exists j \text{ such that } f_j(y)\geq r
  \text{ for every } y\in K,\\
  \operatorname{Multi}_K(r)
  &\iff
  \forall y\in K,\ \exists j \text{ such that } f_j(y)\geq r .
\end{aligned}
\]
\end{definition}
\end{unverified}

\begin{unverified}
\begin{theorem}[Band comparison on a convex body]
\label{thm:single-multi-band-comparison}
Let $K$ be an $m$-dimensional compact convex body with $m\geq 1$, and let
$f_1,\ldots,f_N:\aff(K)\to\R$ be affine functions satisfying
$f_j\leq 1$ on~$K$. Then, for every $r$ with $0<r\leq 1$,
\[
  \operatorname{Single}_K(r)
  \implies
  \operatorname{Multi}_K(r)
  \implies
  \operatorname{Single}_K\bigl(1-(m+1)(1-r)\bigr).
\]
\end{theorem}
\end{unverified}

\begin{proof}
The first implication is immediate from
Definition~\ref{def:single-multi-band-guards}.

For the second implication, assume $\operatorname{Multi}_K(r)$.
Choose $z\in K$ as in Fact~\ref{fact:minkowski-asymmetry-center}. By
$\operatorname{Multi}_K(r)$, choose $j_0$ such that $f_{j_0}(z)\geq r$.
Let $g=1-f_{j_0}$. Then $g$ is affine, $g\geq 0$ on~$K$, and
$g(z)\leq 1-r$.

For any $y\in K$, put
\[
  y' = z-\frac{1}{m}(y-z).
\]
Then $y'\in K$ and
\[
  z = \frac{1}{m+1}y + \frac{m}{m+1}y'.
\]
By affinity and nonnegativity of~$g$ on~$K$,
\[
  g(z)
  = \frac{1}{m+1}g(y) + \frac{m}{m+1}g(y')
  \geq \frac{1}{m+1}g(y).
\]
Hence $g(y)\leq (m+1)g(z)\leq (m+1)(1-r)$, so
\[
  f_{j_0}(y)
  = 1-g(y)
  \geq 1-(m+1)(1-r).
\]
Since this holds for every $y\in K$,
$\operatorname{Single}_K(1-(m+1)(1-r))$ holds.
\end{proof}

\begin{unverified}
\begin{corollary}[Band comparison for a 4D polytope facet]
\label{cor:four-dimensional-facet-band-comparison}
Let $P=\{x\in\R^4:\langle a_k,x\rangle\leq 1,\ k\in I\}$ and
let
\[
  F_i=P\cap\{x:\langle a_i,x\rangle=1\}.
\]
For retained facets $j\in I\setminus\{i\}$, set
$f_j(y)=\langle a_j,y\rangle$ on~$F_i$.
Assume that $F_i$ is compact and has dimension $m$ with $1\leq m\leq 3$.
Then for every $r$ with $0<r\leq 1$,
\[
  \operatorname{Single}_{F_i}(r)
  \implies
  \operatorname{Multi}_{F_i}(r)
  \implies
  \operatorname{Single}_{F_i}(4r-3).
\]
\end{corollary}
\end{unverified}

\begin{proof}
Apply Theorem~\ref{thm:single-multi-band-comparison} with
$m=\dim F_i$. For a genuine facet of a 4D polytope, $m=3$; the same argument
also covers lower-dimensional nonempty faces with $1\leq m\leq 3$.
Since $f_j\leq 1$ on $F_i$ for every retained facet~$j$,
the hypotheses hold. The theorem gives
\[
  \operatorname{Multi}_{F_i}(r)
  \implies
  \operatorname{Single}_{F_i}\bigl(1-(m+1)(1-r)\bigr).
\]
Because $m+1\leq 4$,
\[
  1-(m+1)(1-r)\geq 1-4(1-r)=4r-3.
\]
Thus $\operatorname{Single}_{F_i}(4r-3)$ follows.
\end{proof}

\begin{unverified}
\begin{theorem}[Multi-band guard removal bound]
\label{thm:multi-band-guard-removal-bound}
Let
\[
  P=\{x\in\R^4:\langle a_k,x\rangle\leq 1,\ k\in I\}
\]
and fix $i\in I$. Put $R=I\setminus\{i\}$,
\[
  P'=\{x\in\R^4:\langle a_k,x\rangle\leq 1,\ k\in R\}
\]
and
\[
  F_i=P\cap\{x:\langle a_i,x\rangle=1\}.
\]
Assume $F_i$ is nonempty. If, for some $r$ with $0<r\leq 1$,
\[
  \forall y\in F_i,\ \exists j\in R
  \quad\text{such that}\quad
  \langle a_j,y\rangle\geq r,
\]
then
\[
  P\subseteq P'\subseteq r^{-1}P.
\]
\end{theorem}
\end{unverified}

\begin{proof}
The inclusion $P\subseteq P'$ is immediate. For the second inclusion, take
$x\in P'$. If $\langle a_i,x\rangle\leq 1$, then $x\in P$ and hence
$x\in r^{-1}P$ because $r\leq 1$. Otherwise put
\[
  \alpha=\langle a_i,x\rangle>1,
  \qquad
  y=x/\alpha.
\]
Then $\langle a_i,y\rangle=1$, and for every $k\in R$,
\[
  \langle a_k,y\rangle
  =\frac{\langle a_k,x\rangle}{\alpha}
  \leq \frac{1}{\alpha}<1,
\]
so $y\in F_i$. Choose $j\in R$ with
$\langle a_j,y\rangle\geq r$. Since $x\in P'$,
\[
  \alpha\,\langle a_j,y\rangle
  =\langle a_j,x\rangle
  \leq 1.
\]
Thus $\alpha\leq r^{-1}$. Therefore
\[
  \langle a_i,x/r^{-1}\rangle\leq 1.
\]
For $k\in R$,
\[
  \langle a_k,x/r^{-1}\rangle
  = r\,\langle a_k,x\rangle
  \leq r\leq 1.
\]
Hence $x/r^{-1}\in P$, so $x\in r^{-1}P$.
\end{proof}

\begin{unverified}
\begin{corollary}[Distortion bounds from a facet-removal containment bound]
\label{cor:facet-removal-capacity-volume-sys-bounds}
In the setting of Lemma~\ref{lem:hrep-facet-removal-containment}, assume
that $P$ and $P_R$ are convex bodies in $\R^4$ and write
$\lambda=1+\delta$. Then
\[
  1
  \leq
  \frac{c_{\mathrm{EHZ}}(P_R)}{c_{\mathrm{EHZ}}(P)}
  \leq
  \lambda^2,
\]
\[
  1
  \leq
  \frac{\vol(P_R)}{\vol(P)}
  \leq
  \lambda^4,
\]
and, for the systolic ratio
$\sys(K)=c_{\mathrm{EHZ}}(K)^2/(2\vol(K))$ from
Definition~\ref{def:systolic-ratio},
\[
  \lambda^{-4}
  \leq
  \frac{\sys(P_R)}{\sys(P)}
  \leq
  \lambda^4.
\]
\end{corollary}
\end{unverified}

\begin{proof}
By Lemma~\ref{lem:hrep-facet-removal-containment},
$P\subseteq P_R\subseteq \lambda P$. Monotonicity
(Theorem~\ref{thm:ehz-monotonicity}) and conformality
(Theorem~\ref{thm:conformality}) give
\[
  c_{\mathrm{EHZ}}(P)
  \leq c_{\mathrm{EHZ}}(P_R)
  \leq c_{\mathrm{EHZ}}(\lambda P)
  = \lambda^2 c_{\mathrm{EHZ}}(P).
\]
Lebesgue measure is monotone under inclusion and degree-$4$ homogeneous under
scaling, hence
\[
  \vol(P)\leq \vol(P_R)\leq \vol(\lambda P)=\lambda^4\vol(P).
\]
Let
\[
  r_c=\frac{c_{\mathrm{EHZ}}(P_R)}{c_{\mathrm{EHZ}}(P)}
  \quad\text{and}\quad
  r_v=\frac{\vol(P_R)}{\vol(P)}.
\]
The previous bounds say $1\leq r_c\leq\lambda^2$ and
$1\leq r_v\leq\lambda^4$. Therefore
\[
  \frac{\sys(P_R)}{\sys(P)}=\frac{r_c^2}{r_v}.
\]
The lower bound follows from $r_c^2\geq 1$ and $r_v\leq\lambda^4$.
The upper bound follows from $r_c^2\leq\lambda^4$ and $r_v\geq 1$.
\end{proof}

\begin{unverified}
\begin{remark}[Experiment contract for near-redundant facet removal]
\label{rem:near-redundant-facet-removal-experiment-contract}
The JSONL field
\[
  \texttt{near\_redundant\_facet\_removal\_delta\_bound}
\]
in \texttt{experiments/dev-f64-capacity} is intended to satisfy the hypothesis of
Lemma~\ref{lem:hrep-facet-removal-containment}: for every removed original facet
$j$ and every point $x$ in the simplified polytope $P_R$,
\[
  \langle a_j,x\rangle \leq 1+\texttt{delta\_bound}.
\]
If that inequality is a valid nonnegative bound, then the reported
\[
\begin{gathered}
  \texttt{capacity\_ratio\_upper\_bound},\quad
  \texttt{volume\_ratio\_upper\_bound},\\
  \texttt{sys\_ratio\_lower\_bound},\quad
  \texttt{sys\_ratio\_upper\_bound}
\end{gathered}
\]
are exactly the corresponding factors in
Corollary~\ref{cor:facet-removal-capacity-volume-sys-bounds}.

The current experiment computes \texttt{delta\_bound} with ordinary f64
arithmetic. That is useful as a trusted-enough empirical diagnostic, but it is
not an outward-rounded certificate. A certificate-strength implementation
would need an exact audit or directed-rounding interval computation for the
maximal removed-facet violation.
\end{remark}
\end{unverified}
