\subsection{Primal-Dual Equivalence}\label{subsec:primal-dual}

% QC: Clarke's Dual Action Principle is classical for smooth convex bodies
%   (Clarke 1979, 1981). The polytope case requires care with the gauge
%   function (not smooth) and the dual functional (integral of h_K^2).
%   Literature on this is sparse and often omits details. This writeup
%   is self-contained for a master's thesis audience.
% Downstream: establishes that minimizing action over Reeb orbits is
%   equivalent to minimizing I_K over centered curves. The dual problem
%   is easier to work with for the simplification algorithm.

We work with absolutely continuous curves
$z \in W^{1,2}([0, T], \mathbb{R}^4)$
(the Sobolev space $H^1$, which embeds into $C^0$).
% QC: W^{1,2} functions on a compact interval are absolutely continuous
%   and continuous. No need for a separate C^0 assumption.

We use the gauge function~$g_K$ (Definition~\ref{def:gauge-function}),
the support function~$h_K$ (Definition~\ref{def:support-function}),
and the Hamiltonian $H = g_K^2$ (Definition~\ref{def:hamiltonian}).

\begin{lemma}[Fenchel duality]\label{lem:fenchel-duality}
% QC: Legendre-Fenchel conjugate of g_K^2 is (1/4)h_K^2.
%   The pointwise inequality is Young's inequality for this pair.
%   Equality characterizes the subgradient relation.
% Downstream: integrated along curves, this links the primal
%   (action + gauge constraint) to the dual (I_K functional).
For any $x, y \in \mathbb{R}^4$:
\begin{equation}\label{eq:fenchel-inequality}
  g_K^2(x) + \tfrac{1}{4} h_K^2(y) \geq \langle x, y \rangle,
\end{equation}
with equality if and only if
$y \in \partial g_K^2(x)$ if and only if
$x \in \partial(\tfrac{1}{4} h_K^2)(y)$.
\end{lemma}

\begin{definition}[Primal problem]\label{def:primal-problem}
% QC: restatement of the EHZ capacity minimization, using H = g_K^2.
%   The Hamiltonian formulation (differential inclusion with subdifferential)
%   is equivalent to the Reeb orbit formulation on polytopes.
The \emph{primal problem} is:
\begin{equation}\label{eq:primal-problem}
  c_{\mathrm{EHZ}}(K) = \min \Bigl\{ A(\gamma) :
  \gamma \text{ is a closed characteristic on } \partial K \Bigr\}.
\end{equation}
In the Hamiltonian formulation, a closed characteristic $\gamma$ on $\partial K$
satisfies $g_K^2(\gamma(t)) \equiv 1$ and
$-J_0\dot{\gamma}(t) \in \partial g_K^2(\gamma(t))$ a.e.,
and its action equals its period: $A(\gamma) = T$.
\end{definition}

\begin{definition}[Dual functional and dual problem]\label{def:dual-problem}
% QC: I_K depends only on the velocity dz/dt, not on z itself.
%   This is the key advantage of the dual formulation: no position constraint
%   (gamma on boundary K) and no differential inclusion constraint.
%   The centering condition removes the translation ambiguity in going
%   from gamma to z = gamma - center(gamma).
The \emph{dual functional} is
\[
  I_K(z) = \frac{1}{4} \int_0^T h_K^2\bigl( -J_0 \dot{z}(t) \bigr) \, dt.
\]
The \emph{dual problem} is:
\begin{equation}\label{eq:dual-problem}
  \min \Bigl\{ I_K(z) : z \in W^{1,2}([0,T], \mathbb{R}^4),\;
  z(0) = z(T),\;
  A(z) = T,\;
  \frac{1}{T} \int_0^T z(t) \, dt = 0,\;
  T > 0 \Bigr\}.
\end{equation}
The constraint $\frac{1}{T} \int z = 0$ is the
\emph{centering condition}; it removes the translation
ambiguity (since $I_K$ depends only on $\dot{z}$, not on~$z$).
\end{definition}

\begin{theorem}[Primal-dual equivalence (Clarke's Dual Action Principle)]%
\label{thm:primal-dual-equivalence}
% QC: the critical-point correspondence is the core content.
%   Existence of minimizers is a separate fact (thm:orbit-existence
%   for the primal; compactness + l.s.c. for the dual).
%   The equivalence proof does not use existence of minimizers.
% Downstream: allows us to work in the dual formulation, where the
%   simplification algorithm (subsec:simplification) is natural.
Let $K$ be a convex body. Critical points of the primal and
dual problems correspond bijectively via translation:
if $\gamma$ is a primal critical point (Hamiltonian orbit on
$\partial K$), then
$z = \gamma - \frac{1}{T} \int_0^T \gamma(t) \, dt$
is a dual critical point, and conversely.
At every corresponding pair, the values are equal:
\begin{equation}\label{eq:primal-dual-values}
  I_K(z) = T = A(\gamma).
\end{equation}
Consequently, $\min_\gamma A(\gamma) = \min_z I_K(z) = c_{\mathrm{EHZ}}(K)$.
\end{theorem}

\emph{Step~1: Fenchel equality for Hamiltonian orbits.}
Let $\gamma$ be a Hamiltonian orbit on $\partial K$ with period~$T$,
so $g_K^2(\gamma) \equiv 1$ and
$-J_0\dot{\gamma} \in \partial g_K^2(\gamma)$ a.e.
The subgradient condition
$-J_0\dot{\gamma} \in \partial g_K^2(\gamma)$
is exactly the equality case of the Fenchel
inequality~\eqref{eq:fenchel-inequality}, so pointwise:
\[
  g_K^2(\gamma(t)) + \tfrac{1}{4} h_K^2(-J_0\dot{\gamma}(t))
  = \langle \gamma(t), -J_0\dot{\gamma}(t) \rangle.
\]
Integrate over $[0, T]$:
\begin{equation}\label{eq:fenchel-integrated}
  \underbrace{\int_0^T g_K^2(\gamma) \, dt}_{= T}
  + \underbrace{\frac{1}{4}\int_0^T h_K^2(-J_0\dot{\gamma}) \, dt}_{= I_K(\gamma)}
  = \underbrace{\int_0^T \langle \gamma, -J_0\dot{\gamma} \rangle \, dt}_{= 2A(\gamma) = 2T}.
\end{equation}
Hence $I_K(\gamma) = T = A(\gamma)$.

\emph{Step~2: Euler--Lagrange equation for the dual problem.}
The dual functional $I_K(z) = \frac{1}{4}\int_0^T h_K^2(-J_0\dot{z}) \, dt$
depends only on $\dot{z}$, not on~$z$ directly.
We seek critical points of $I_K$ under the constraints:
\begin{itemize}
\item closure: $z(0) = z(T)$,
\item action: $\int_0^T \langle -J_0\dot{z}, z \rangle \, dt = 2T$
      (equivalently, $A(z) = T$),
\item centering: $\int_0^T z(t) \, dt = 0$.
\end{itemize}

We compute the critical points using the calculus of variations with Lagrange multipliers.
Form the Lagrangian incorporating the constraints:
\[
  L(z, \dot{z}, t) = \tfrac{1}{4}h_K^2(-J_0\dot{z}) + \alpha\langle -J_0\dot{z}, z\rangle + \langle \mu, z\rangle,
\]
where $\alpha \in \mathbb{R}$ is the multiplier for the action constraint $\int_0^T \langle -J_0\dot{z}, z \rangle \, dt = 2T$,
and $\mu \in \mathbb{R}^4$ is the multiplier for the centering constraint $\int_0^T z \, dt = 0$.

The Euler--Lagrange equation is
$\frac{d}{dt}[\partial L / \partial \dot{z}] = \partial L / \partial z$.
The right-hand side comes solely from the multiplier terms
(since $\tfrac{1}{4}h_K^2(-J_0\dot{z})$ is independent of~$z$):
$\partial L / \partial z = -\alpha J_0 \dot{z} + \mu$.
For the left-hand side, let $p(t) \in \partial(\tfrac{1}{4}h_K^2)(-J_0\dot{z}(t))$.
The chain rule for $f \circ A$ with $A = -J_0$ gives
$(-J_0)^\top p = J_0 p$ (using $J_0^\top = -J_0$).
Similarly $\partial / \partial \dot{z}\, [\alpha\langle -J_0\dot{z}, z\rangle] = \alpha J_0 z$.
Hence $\partial L / \partial \dot{z} = J_0 p(t) + \alpha J_0 z(t)$, and
\[
  \frac{d}{dt}[J_0 p + \alpha J_0 z]
  = J_0 \dot{p} + \alpha J_0 \dot{z}.
\]
Equating: $J_0 \dot{p} + \alpha J_0 \dot{z} = -\alpha J_0 \dot{z} + \mu$,
so $J_0 \dot{p} = -2\alpha J_0 \dot{z} + \mu$.

Integrating over $[0,T]$ and using periodicity of~$p$ and~$z$:
\[
  J_0\underbrace{[p(T) - p(0)]}_{= 0}
  = -2\alpha J_0 \underbrace{[z(T) - z(0)]}_{= 0} + T\mu,
\]
hence $\mu = 0$. Thus $\dot{p} = -2\alpha \dot{z}$ a.e.,
so $p(t) = c - 2\alpha\, z(t)$ for a constant $c \in \mathbb{R}^4$.
Since $p(t) \in \partial(\tfrac{1}{4}h_K^2)(-J_0\dot{z}(t))$:
\[
  c - 2\alpha\, z(t)
  \in \partial\bigl(\tfrac{1}{4}h_K^2\bigr)\bigl(-J_0\dot{z}(t)\bigr)
  \quad \text{a.e.}
\]
Writing $\nu = -2\alpha$ and $b = c$:
\begin{equation}\label{eq:dual-euler-lagrange}
  \nu z(t) + b \in
  \partial\bigl(\tfrac{1}{4} h_K^2\bigr)\bigl(-J_0\dot{z}(t)\bigr)
  \quad \text{a.e.}
\end{equation}
for some $\nu \in \mathbb{R}$ and constant $b \in \mathbb{R}^4$.

\emph{Step~3: The $\nu = 1$ argument and correspondence.}
Define $\gamma(t) = \nu z(t) + b$. The Euler--Lagrange
condition~\eqref{eq:dual-euler-lagrange} gives
$\gamma(t) \in \partial(\tfrac{1}{4}h_K^2)(-J_0\dot{\gamma}/\nu)$ a.e.
By Fenchel duality, this is equivalent to
$-J_0\dot{\gamma}/\nu \in \partial g_K^2(\gamma)$ a.e.

\subsection{Simplification Algorithm}\label{subsec:simplification}

% QC: the heart of the proof. Takes an arbitrary dual minimizer z
%   and constructs a simple dual minimizer z^* via 5 steps. Each step
%   is designed to preserve or improve the optimality while adding
%   structure. The final compactness argument (step 5) ensures we
%   can take a limit that is both simple and optimal.

We now prove the simple minimizer theorem
(Theorem~\ref{thm:simple-minimizer}):
\emph{for every polytope~$K$, there exists a minimum-action Reeb orbit
that is simple} (piecewise constant velocity, pure facet Reeb vectors,
each facet used at most once).

\begin{proof}[Proof of Theorem~\ref{thm:simple-minimizer}]
A minimum-action closed characteristic on~$\partial K$ exists
(Theorem~\ref{thm:orbit-existence}): the infimum of the action over
all closed characteristics is attained and positive.
By the primal-dual equivalence (Theorem~\ref{thm:primal-dual-equivalence}),
any such primal minimizer~$\gamma$ corresponds to a dual minimizer
$z = \gamma - \mathrm{center}(\gamma)$ with $I_K(z) = T = A(\gamma)$.
We will construct a simple dual minimizer~$z^*$, which then gives a
simple primal minimizer via $\gamma^* = z^* + b$.

Let $z$ be an arbitrary dual minimizer with period~$T > 0$. By the
primal-dual equivalence, $A(z) = T = I_K(z)$. We construct a sequence
of curves $z \to z' \to z'' \to z''' \to z^{(4)} \to z^*$, where each
arrow represents one simplification step. The final curve~$z^*$ is
simple (piecewise linear with pure Reeb vectors as velocities, each
Reeb vector used at most once) and satisfies $A(z^*) = T = I_K(z^*)$,
hence is also a minimizer.

\paragraph{Step 0: Setup.}
% QC: z is any dual minimizer. We know A(z) = T = I_K(z) from the
%   primal-dual equivalence. The goal is to perturb z into a simple
%   curve while preserving optimality.
We have $z \in W^{1,2} \cap C^0$ with $z(0) = z(T)$,
$\frac{1}{T} \int_0^T z = 0$, and $A(z) = T = I_K(z)$.

\paragraph{Step 1: Piecewise linear approximation.}
By the primal-dual correspondence (Theorem~\ref{thm:primal-dual-equivalence}),
$z$ corresponds to a primal minimizer $\gamma$ on $\partial K$.
Since $\gamma$ is a closed characteristic on $\partial K$, we have
$\dot{\gamma}(t) \in J_0 N_K(\gamma(t))$ a.e., which for polytopes equals
$\mathrm{conv}\{R_1,\ldots,R_F\}$ (the convex hull of facet Reeb vectors).
By the correspondence $z = \gamma - \mathrm{center}(\gamma)$, we have
$\dot{z}(t) \in \mathrm{conv}\{R_1,\ldots,R_F\}$ a.e.

% [AGENT CONFIDENCE 90%: Claim statement verified against HK2017 Lemma 4.2 (lines 275-293

\textbf{Claim (Piecewise affine approximation):}
For any $\varepsilon > 0$, there exists a piecewise affine curve $z' : [0,T] \to \mathbb{R}^4$ such that:
\begin{itemize}
\item $z'(0) = z'(T)$ (closed),
\item $\|z - z'\|_{W^{1,2}} < \varepsilon$ (approximation in Sobolev norm),
\item $\dot{z}'(t) \in \mathrm{conv}\{R_1,\ldots,R_F\}$ on each affine piece,
\item $|A(z') - A(z)| < \varepsilon$ and $|I_K(z') - I_K(z)| < \varepsilon$
      (by continuity of $A$ and $I_K$ in $W^{1,2}$).
\end{itemize}
The centering condition $\frac{1}{T}\int_0^T z' = 0$ is ensured by translating $z'$.

This is a standard density result for Sobolev spaces (HK2017 Lemma 4.2 gives the construction with the cone constraint).

\paragraph{Step 2: Split into pure Reeb vectors.}
After Step~1, each velocity segment $\dot{z}'(t)$ is a convex combination
of facet Reeb vectors. We replace each such segment by a concatenation
of segments with \emph{pure} facet Reeb vectors (not convex combinations).

% [AGENT CONFIDENCE 85%: Claim structure matches HK2017 Lemma 4.4 (lines 317-355) and
%  references Proposition 4.3 (lines 301-315) for action identity. Agent verified the

\textbf{Claim (Splitting convex combinations):}
If $\dot{z}'(t) = \sum_{i=1}^k a_i(t) R_i$ with $\sum a_i(t) = 1$ on interval $I$,
replace $I$ by $k$ subintervals of lengths $\Delta t_i = \int_I a_i(t)\,dt$,
each with pure velocity $R_i$.
Total time preserved: $\sum \Delta t_i = \int_I 1\,dt = |I|$.

The time ordering affects action. Reversing the order reverses the action change (sign flip).
Choose the ordering that makes $A(z'') \geq A(z')$.

(HK2017 Lemma 4.4 proves the splitting construction; Proposition 4.3 gives the action identity for piecewise constant velocities.)

After splitting, the dual functional equals the period \emph{exactly}:

The identity $I_K(z'') = T$ follows from the Reeb normalization:
for $\dot{z}(t) = R_i = \frac{2}{h_i} J_0 n_i$, we have
\begin{align*}
  h_K^2(-J_0 \dot{z}(t))
  &= h_K^2\Bigl( -J_0 \cdot \frac{2}{h_i} J_0 n_i \Bigr)
  = h_K^2\Bigl( \frac{2}{h_i} n_i \Bigr) \\
  &= \Bigl( \frac{2}{h_i} \Bigr)^2 h_K^2(n_i)
  = \frac{4}{h_i^2} \cdot h_i^2
  = 4,
\end{align*}
hence $I_K(z'') = \frac{1}{4} \int_0^T 4 \, dt = T$.

\paragraph{Step 3: Merge repeated Reeb vectors.}
After Step~2, the velocity $\dot{z}''$ takes values in the finite set
$\{R_1,\ldots,R_F\}$ (pure facet Reeb vectors).
If the same Reeb vector $R_i$ appears in two disjoint time intervals,
we merge them into a single contiguous segment.

% [AGENT-WRITTEN CLAIM, UNREVIEWED BY JÖRN]
% [AGENT CONFIDENCE 85%: Claim structure matches HK2017 Lemma 4.5 (lines 356-397). Agent
%  verified the rearrangement mechanism (forward/backward merge), the action identity for
%  piecewise constant velocities, and the sign-flip property. The conclusion (choose the
%  merge that preserves or increases action) is correctly stated. NEEDS PROOF: either inline

\textbf{Claim (Merging repeated velocities):}
Consider a pattern where $R_A$ appears in intervals $I_r$ and $I_s$ with $r < s$,
separated by intervals $I_{r+1}, \ldots, I_{s-1}$ with different velocities.
We can rearrange to move the two $R_A$ segments adjacent:
\begin{itemize}
\item \textbf{Forward merge:} Place both $R_A$ segments at position $r$,
      then $I_{r+1}, \ldots, I_{s-1}$ afterward.
\item \textbf{Backward merge:} Place $I_{r+1}, \ldots, I_{s-1}$ first,
      then both $R_A$ segments.
\end{itemize}
These differ by a reversal of the intermediate portion. By the
piecewise-constant velocity action identity
\[
  2A(z) = \sum_{j<i} |I_i| \, |I_j| \, \omega_0(\dot{z}|_{I_i}, \dot{z}|_{I_j}),
\]
reversing segments flips the sign of their contribution to action. Hence:
\[
  \Delta A_{\text{fwd}} = -\Delta A_{\text{bwd}}.
\]
One option has $\Delta A \geq 0$ (or both equal). Choose that one.

(HK2017 Lemma 4.5 proves this rearrangement preserves closure and allows action control.)

The dual functional $I_K$ is unchanged: it depends only on
$h_K^2(-J_0\dot{z})$ integrated over time, which depends on the
multiset of velocities and their durations, not their ordering.

Iterating for all repeated Reeb vectors yields $z'''$ with:
\begin{itemize}
\item $A(z''') \geq A(z'')$ (each merge preserves or increases action),
\item $I_K(z''') = I_K(z'') = T$ (rearrangement preserves $I_K$),
\item each Reeb vector appears in at most one contiguous time interval.
\end{itemize}

\paragraph{Step 4: Rescale to restore normalization.}
After Steps 2--3, the action may have changed slightly:
$A(z''') \geq A(z'') \geq A(z') \geq T - \varepsilon$.
We restore the normalization $A(z) = T$ by scaling all time intervals
proportionally.

Define the scaling factor
\[
  \beta = \frac{T}{A(z''')} \in (0, 1 + O(\varepsilon/T)].
\]
Construct $z^{(4)}$ by \emph{multiplying each time interval by $\beta$}:
\begin{itemize}
\item Original: velocity $R_i$ on interval of length $\Delta t_i$
\item Scaled: velocity $R_i$ on interval of length $\beta \cdot \Delta t_i$
\end{itemize}
The velocities remain unchanged (still pure Reeb vectors in direction and magnitude).

\textbf{Consequences:}
\begin{align*}
  T^{(4)} &= \beta \sum \Delta t_i = \beta \cdot T = \frac{T^2}{A(z''')}, \\
  A(z^{(4)}) &= \beta^2 \cdot A(z''')
    \quad\text{(action scales as [time]$^2$ when velocities constant)} \\
  &= \left(\frac{T}{A(z''')}\right)^2 \cdot A(z''') = \frac{T^2}{A(z''')} = T^{(4)}, \\
  I_K(z^{(4)}) &= \beta \cdot I_K(z''')
    \quad\text{($I_K = \frac{1}{4}\int h_K^2(-J_0\dot{z})\,dt$ is linear in time)} \\
  &= \frac{T}{A(z''')} \cdot T = \frac{T^2}{A(z''')}.
\end{align*}
Hence $A(z^{(4)}) = T^{(4)}$ (normalization restored) and
$I_K(z^{(4)}) = T^{(4)}$.

Since $z$ was a dual minimizer with $I_K(z) = T$, feasibility gives
$I_K(z^{(4)}) \geq T$, hence $T^2/A(z''') \geq T$, so $A(z''') \leq T$.
Combined with $A(z''') \geq T - \varepsilon$, as $\varepsilon \to 0$
we have $A(z''') \to T$, so $\beta \to 1$ and $T^{(4)} \to T$.

\paragraph{Step 5: Compactness and limit.}
A simple curve is encoded by finite data:
\begin{itemize}
\item A permutation $\sigma \in S_F$ (facet visit order),

\end{itemize}

Consider the space $\mathcal{E}^1$ of piecewise affine curves
on $[0,T^{(4)}]$ with velocities in $\{R_1,\ldots,R_F\}$
(each $R_i$ appearing at most once).
Define a map
\[
  \Phi : \mathcal{E}^1 \to S_F \times \Delta^{F-1},
  \quad z' \mapsto \left(\sigma, \left(\frac{t_1}{T^{(4)}},\ldots,\frac{t_F}{T^{(4)}}\right)\right),
\]
where $\Delta^{F-1} = \{(s_1,\ldots,s_F) : s_i \geq 0, \sum s_i = 1\}$
is the standard $(F-1)$-simplex.

The image $\mathrm{Im}(\Phi) \subset S_F \times \Delta^{F-1}$ is compact
(finite set times a compact set).
The sequence $\{z^{(4)}_n\}$ (from Step~4 with $\varepsilon_n \to 0$) lies in $\mathcal{E}^1$.

After passing to a subsequence:
\begin{itemize}
\item The permutation $\sigma^n = \sigma$ is constant,
\item The normalized durations converge:
      $(t^n_1/T^{(4)}_n,\ldots,t^n_F/T^{(4)}_n) \to (s^{\infty}_1,\ldots,s^{\infty}_F)$ in $\Delta^{F-1}$,
\item The periods $T^{(4)}_n \to T$ (from Step~4 analysis).
\end{itemize}

Define $z^*$ as the piecewise affine curve with ordering $\sigma$,
velocities $\{R_{\sigma(i)}\}$, and time durations $t^{\infty}_i = s^{\infty}_i \cdot T$.

Then $\|z^{(4)}_n - z^*\|_{W^{1,2}} \to 0$ (convergence in $W^{1,2}$):
The velocity $\dot{z}^{(4)}_n$ matches $\dot{z}^*$ on most of $[0,T]$,
with the measure of mismatch tending to zero as
$T^{(4)}_n \to T$ and $t^n_i/T^{(4)}_n \to s^{\infty}_i$.

By continuity of $A$ and $I_K$ in $W^{1,2}$:
\[
  A(z^*) = \lim_{n\to\infty} A(z^{(4)}_n) = T,
  \qquad
  I_K(z^*) = \lim_{n\to\infty} I_K(z^{(4)}_n) = T.
\]
Since $z$ was a dual minimizer with $I_K(z) = T$, and $I_K(z^*) = T$,
the curve $z^*$ is also a dual minimizer. It is simple by construction
(piecewise affine, pure Reeb velocities, each facet used at most once).

\paragraph{Conclusion.}
By Theorem~\ref{thm:primal-dual-equivalence}, the dual minimizer~$z^*$
corresponds to a primal minimizer $\gamma = z^* + b$ for some constant
$b \in \mathbb{R}^4$. Since $z^*$ is simple (piecewise linear with pure
Reeb vectors, each used at most once), $\gamma$ is also simple
(Definition~\ref{def:simple-reeb-orbit}). This completes the proof.
\end{proof}
