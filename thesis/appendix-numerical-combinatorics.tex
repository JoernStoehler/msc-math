The algorithms in Section~\ref{sec:algorithms} operate on a polytope
$K = K(n, h) = \{x \in \mathbb{R}^4 : \langle n_i, x \rangle \leq h_i,
\; i = 1, \ldots, F\}$.
In practice, the coordinates are represented in
\texttt{f64} (IEEE~754 double precision).
This section addresses the combinatorial structure of polytopes
under this representation.

The key idea is to avoid numerical predicates entirely
for discrete combinatorial decisions.
We represent polytopes with exact rational coordinates
($n_i \in \mathbb{Q}^4$, $h_i \in \mathbb{Q}_{>0}$),
compute combinatorial data exactly over~$\mathbb{Q}$,
and convert to \texttt{f64} only for continuous computation
(e.g.\ the SVD step of the KKT solver).
The exact rational polytope is the source of truth
for all combinatorial decisions;
the \texttt{f64} polytope is a derived quantity
for fast numerical computation.

\begin{definition}[Combinatorial data]%
\label{def:combinatorial-data}
Let $K = K(n, h)$ be a simple polytope with
$F$~facets in~$\mathbb{R}^4$,
where the normals $n_i$ are nonzero (not necessarily unit).
Its \emph{combinatorial data} is $C(K) = (\mathcal{V}, D)$
where:
\begin{enumerate}
\item \emph{Vertex descriptors.}
  $\mathcal{V} = \{S_1, \ldots, S_V\}$
  where $S_j = \{i : v_j \in F_i\}$ is the set of
  facets through vertex~$v_j$.
  By simplicity, $|S_j| = 4$ for each~$j$.
\item \emph{Facet adjacency.}
  Facets $i$ and~$k$ are
  \emph{adjacent} if $\{i, k\} \subset S_j$ for some~$j$,
  i.e.\ they share a vertex.
\item \emph{Sign pattern.}
  $D(i, k) = \operatorname{sign}\bigl(
  \omega_0(n_i, n_k)\bigr) \in \{+, -, 0\}$
  for each adjacent pair $(i, k)$.
\end{enumerate}
\end{definition}

\begin{remark}[Scope: simple polytopes]%
\label{rem:simple-scope}
We restrict to simple polytopes throughout.
This covers all polytopes arising in our experiments:
random polytopes are almost surely simple,
Lagrangian products are always simple,
and polytopes from future optimisation procedures
(e.g.\ gradient descent on the systolic ratio)
can be perturbed to simple ones if needed.
\end{remark}

\begin{definition}[Rational polytope]%
\label{def:rational-polytope}
A \emph{rational polytope} is $K = K(n, h)$ with
normals $n_i \in \mathbb{Q}^4 \setminus \{0\}$
and heights $h_i \in \mathbb{Q}_{>0}$.
The normals are \emph{not} required to be unit vectors.

Given a 4-element subset $S \subset \{1, \ldots, F\}$,
let $N_S$ denote the $4 \times 4$ matrix whose rows
are $n_i^\top$ for $i \in S$.
The \emph{vertex at~$S$} is $v_S = N_S^{-1}\, h_S$
(computed via Cramer's rule, exact over~$\mathbb{Q}$)
when $\det(N_S) \neq 0$.
The subset $S$ is a \emph{vertex descriptor}
of~$K$ if $\det(N_S) \neq 0$ and all
\emph{non-incidence gaps}
$g^i_S \coloneqq h_i - \langle n_i, v_S \rangle > 0$
for $i \notin S$ are strictly positive.
\end{definition}

\begin{definition}[Margins]%
\label{def:margins}
Let $K = K(n, h)$ be a simple rational polytope
with combinatorial data
$C(K) = (\mathcal{V}, D)$.
The \emph{margins} of~$K$ are:
\begin{enumerate}
\item $\gamma_{\min} \coloneqq
  \min_{S_j \in \mathcal{V}} \;\min_{i \notin S_j}\;
  g^i_{S_j}$
  \quad (smallest non-incidence gap),
\item $\delta_{\min} \coloneqq
  \min_{S_j \in \mathcal{V}} \;|\!\det(N_{S_j})|$
  \quad (smallest vertex-matrix determinant),
\item $\omega_{\min} \coloneqq
  \min_{\substack{(i,k) \in \operatorname{adj}(\mathcal{V})
  \\ D(i,k) \neq 0}} |\omega_0(n_i, n_k)|$
  \quad (smallest nonzero $|\omega_0|$ value).
\end{enumerate}
All three quantities are computed exactly over~$\mathbb{Q}$.
By construction, $\gamma_{\min} > 0$ and
$\delta_{\min} > 0$.
If all adjacent pairs have $D(i,k) = 0$,
then $\omega_{\min}$ is undefined (vacuously satisfied).
\end{definition}

\begin{lemma}[Self-consistency]%
\label{lem:self-consistency}
Every simple rational polytope $K$ is consistent with
its own combinatorial data~$C(K)$.
The margins $\gamma_{\min}$, $\delta_{\min}$, and
$\omega_{\min}$ (when defined) are strictly positive.
\end{lemma}

\begin{proof}
$C(K)$ is defined as the data extracted from~$K$,
so $C(K) = C(K)$.
The margins are positive by the definitions:
$\gamma_{\min} > 0$ because non-incidence gaps define
vertex descriptors,
$\delta_{\min} > 0$ because vertices arise only from
non-singular subsystems,
and $\omega_{\min} > 0$ because the minimum is taken
over pairs with $D(i,k) \neq 0$.
\end{proof}

\begin{remark}[Conversion to \texttt{f64}]%
\label{rem:f64-conversion}
A rational polytope $K = K(n, h)$ is converted to
\texttt{f64} by normalizing each normal to a unit vector:
\[
  \hat n_i = \frac{n_i}{\|n_i\|}, \qquad
  \hat h_i = \frac{h_i}{\|n_i\|}.
\]
The halfspace $\langle n_i, x \rangle \leq h_i$ is
equivalent to
$\langle \hat n_i, x \rangle \leq \hat h_i$.
The normalization and the rational-to-\texttt{f64}
conversion each introduce $O(\varepsilon_{\mathrm{mach}})$
rounding error per coordinate.
\end{remark}

\begin{proposition}[Combinatorial preservation under rounding]%
\label{prop:f64-preservation}
Let $K = K(n, h)$ be a simple rational polytope with
margins $\gamma_{\min}$, $\delta_{\min}$, $\omega_{\min}$.
Let $\hat K = K(\hat n, \hat h)$ be the \texttt{f64}
polytope obtained by normalizing and rounding
(Remark~\ref{rem:f64-conversion}).
If the coordinate-wise rounding error satisfies
$\|n_i / \|n_i\| - \hat n_i\| \leq \varepsilon$
and $|h_i / \|n_i\| - \hat h_i| \leq \varepsilon$
for all~$i$, and $\varepsilon$ is small enough relative
to the margins, then the \texttt{f64} polytope~$\hat K$
has the same combinatorial data as~$K$:
\begin{enumerate}
\item Same vertex descriptors: every vertex of~$K$
  corresponds to a vertex of~$\hat K$ and vice versa.
\item Same adjacency: the facet adjacency graph is identical.
\item Same sign pattern: for every adjacent pair $(i,k)$,
  $\operatorname{sign}(\omega_0(\hat n_i, \hat n_k))
  = D(i,k)$.
\end{enumerate}
\end{proposition}

\begin{remark}[Lagrangian products]%
\label{rem:lagrangian-preservation}
For a Lagrangian product $K = K_q \times_L K_p$,
the normals of~$K_q$ lie in the $q$-plane
$\{(n_1, n_2, 0, 0)\}$ and those of~$K_p$ in the
$p$-plane $\{(0, 0, n_3, n_4)\}$.
For same-type adjacent pairs (both $q$-facets or
both $p$-facets), $\omega_0(n_i, n_k) = 0$ exactly,
because $\omega_0$ vanishes on each Lagrangian plane.
This holds for both the rational and the \texttt{f64}
polytope (the zero is structural, not approximate).
The sign pattern of a Lagrangian product therefore has
$D(i,k) = 0$ for all same-type pairs, and
$D(i,k) \neq 0$ only for cross-type pairs
(one $q$-facet and one $p$-facet).
\end{remark}

\begin{remark}[Application to algorithm soundness]%
\label{rem:interpretation}
The implementation follows this two-level architecture:
\begin{enumerate}
\item \emph{Exact level.}
  The polytope is stored with rational coordinates.
  Combinatorial data~$C = (\mathcal{V}, D)$ and
  margins are computed exactly over~$\mathbb{Q}$
  at construction time.
\item \emph{Rounded level.}
  The \texttt{f64} polytope (unit normals, adjusted heights)
  is derived from the rational polytope.
  The capacity algorithm uses:
  \begin{itemize}
  \item the \emph{exact} combinatorial data for discrete
    decisions (the $(S, \sigma)$ filter:
    which facet subsets to enumerate, and which
    $\omega_0$ sign assignments to consider),
  \item the \emph{rounded} \texttt{f64} coordinates
    for continuous computation
    (the SVD-based KKT solver for~$\beta$, $\eta$, $\nu$).
  \end{itemize}
\end{enumerate}
This separation eliminates numerical predicates
from all combinatorial decisions.
The only remaining source of numerical error is the
continuous KKT computation, which is addressed
in Section~\ref{app:svd-step}.
\end{remark}
