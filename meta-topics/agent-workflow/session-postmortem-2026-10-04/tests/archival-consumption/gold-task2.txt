+def decision(config, primary, now):
+    scheduled = dt.datetime.fromisoformat(config['scheduled_at']).timestamp()
+    if now < scheduled:
+        return 'before_scheduled_time'
+    if now >= config['expires_at']:
+        return 'expired'
+    outcome = primary.get('outcome')
+    if outcome in SUCCESS:
+        return 'primary_succeeded'
+    if outcome in REFUSAL:
+        return 'server_refused'
+    return 'fallback'
