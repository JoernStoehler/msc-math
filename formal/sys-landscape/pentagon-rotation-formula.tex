\subsection{Pentagon Rotation Formula (Draft)}
\label{sec:pentagon-rotation-formula}

This section is a private proof draft for the explicit formula of
\[
  \mathrm{sys}\!\left(P_5 \times_L R(\theta) P_5\right).
\]
It is intended for agent and Jörn review, not yet thesis-facing exposition.

Let
\[
  v_k := \left(
    \cos\left(\frac{2\pi k}{5}\right),
    \sin\left(\frac{2\pi k}{5}\right)
  \right),
  \qquad 0 \le k \le 4,
\]
and let~$P_5$ be the regular pentagon with those vertices. Since rotating both
factors by the same angle is symplectic, the experiment convention used in
`experiments/sys-landscape/pentagon-rotation-formula/main.rs` is symplectically
equivalent to the standard one used here. The HKO special case uses the second
pentagon with vertices at angles $-\pi/2 + 2\pi k/5$, i.e. a rotation by
$-\pi/2$. Modulo the pentagon's $2\pi/5$ rotational symmetry this is the same
midpoint configuration as the present parameter value $\theta=\pi/10$, up to
the sign convention for rotating the second factor.

\begin{unverified}
\paragraph{Conjectural target.}
Let~$P_5 \subset \mathbb{R}^2$ be the regular pentagon with circumradius~$1$
in the angle convention used by the experiment code. For
\[
  K_\theta := P_5 \times_L R(\theta) P_5,
  \qquad 0 \le \theta \le \frac{\pi}{5},
\]
one has
\[
  \mathrm{sys}(K_\theta)
  =
  \frac{5 + 2\sqrt{5}}{10}
  \sec^2\!\bigl(\min(\theta, \tfrac{\pi}{5} - \theta)\bigr).
\]
Equivalently,
\[
  \mathrm{sys}(K_\theta)
  =
  \frac{5 + 2\sqrt{5}}{10}\sec^2(\theta)
  \quad \text{for } 0 \le \theta \le \frac{\pi}{10},
\]
and the second half of the fundamental domain follows by the symmetry
$\theta \mapsto \frac{\pi}{5} - \theta$.
\end{unverified}

\begin{unverified}
\begin{proposition}[The active 2-bounce branch has the claimed secant law]
\label{prop:pentagon-rotation-two-bounce}
Let~$K_\theta := P_5 \times_L R(\theta)P_5$ with
$0 \le \theta \le \frac{\pi}{10}$. Among the 2-bounce trajectories joining a
vertex of~$P_5$ to the opposite edge, the minimizing branch is attained at
\[
  \lambda(\theta)
  =
  \frac{1}{2}
  -
  \frac{\tan(\theta)}{2\tan(\pi/10)},
\]
and its capacity is
\[
  c_{2\text{-bounce}}(K_\theta)
  =
  \left(1+\cos\left(\frac{\pi}{5}\right)\right)^2 \sec(\theta).
\]
Consequently,
\[
  \mathrm{sys}(K_\theta)
  \le
  \frac{5 + 2\sqrt{5}}{10}\sec^2(\theta)
  \qquad
  \text{for } 0 \le \theta \le \frac{\pi}{10}.
\]
\end{proposition}

\begin{proof}
Set
\[
  A := 1 + \cos\left(\frac{\pi}{5}\right),
  \qquad
  B := \sin\left(\frac{\pi}{5}\right).
\]
The HKO classification of non-translatable $2$-bounce trajectories depends only
on the table pentagon, so on the half-domain it is enough to study the family
joining the vertex $v_0$ to the opposite edge $[v_2,v_3]$. In the present
coordinates that edge is vertical, so we parametrize it by
\[
  x_\lambda := \lambda v_2 + (1-\lambda)v_3
  =
  \left(
    -\cos\left(\frac{\pi}{5}\right),
    (2\lambda-1)B
  \right),
  \qquad 0 \le \lambda \le 1.
\]
The corresponding chord vector is
\[
  d_\lambda := x_\lambda - v_0 = (-A,(2\lambda-1)B).
\]

Write
\[
  w_k(\theta) := R(\theta)v_k.
\]
For $0 \le \theta \le \pi/10$, the vector $-d_\lambda$ stays in the normal cone
of $w_0(\theta)$ and $d_\lambda$ stays in the union of the adjacent normal
cones of $w_2(\theta)$ and $w_3(\theta)$. Hence the branch action is
\[
  L_\theta(\lambda)
  =
  \langle -d_\lambda, w_0(\theta)\rangle
  +
  \max\bigl(
    \langle d_\lambda, w_2(\theta)\rangle,
    \langle d_\lambda, w_3(\theta)\rangle
  \bigr).
\]
The support switch is determined by
\[
  \langle d_\lambda, w_2(\theta)-w_3(\theta)\rangle = 0,
\]
which reduces to the affine relation
\[
  A\sin\theta + (2\lambda-1)B\cos\theta = 0.
\]
Thus the switch parameter is
\[
  \lambda(\theta)
  =
  \frac{1}{2} - \frac{\tan(\theta)}{2\tan(\pi/10)}.
\]

For fixed $\theta$, the function $L_\theta$ is affine on the two support cells
$\lambda \le \lambda(\theta)$ and $\lambda \ge \lambda(\theta)$. The routine
trigonometric reduction of the corresponding derivatives is part of the exact
CAS witness from Remark~\ref{rem:pentagon-rotation-cas-witness}, which records
\[
  \frac{d}{d\lambda} L_\theta^{-}
  =
  -\sqrt{5}\sin\left(\theta+\frac{\pi}{10}\right),
  \qquad
  \frac{d}{d\lambda} L_\theta^{+}
  =
  \sqrt{5}\cos\left(\theta+\frac{2\pi}{5}\right).
\]
On $0 \le \theta \le \pi/10$, the left slope is strictly negative and the right
slope is nonnegative, with equality on the right only at the endpoint
$\theta=\pi/10$. Therefore $L_\theta$ decreases up to $\lambda(\theta)$ and
then increases, so $\lambda(\theta)$ is the unique minimizing parameter on this
family.

Substituting the switch relation at the minimizing parameter gives
\[
  d_{\lambda(\theta)} = (-A,-A\tan\theta).
\]
The same exact CAS witness verifies the support values
\[
  h_{R(\theta)P_5}(-d_{\lambda(\theta)}) = A\sec\theta,
  \qquad
  h_{R(\theta)P_5}(d_{\lambda(\theta)})
  =
  A\cos\left(\frac{\pi}{5}\right)\sec\theta,
\]
and therefore
\[
  c_{2\text{-bounce}}(K_\theta)
  =
  A^2\sec\theta.
\]

Finally,
\[
  \operatorname{vol}(K_\theta)=\operatorname{area}(P_5)^2,
  \qquad
  \operatorname{area}(P_5)=\frac{5}{2}\sin\left(\frac{2\pi}{5}\right),
\]
so
\[
  \mathrm{sys}(K_\theta)
  \le
  \frac{A^4}{2\operatorname{area}(P_5)^2}\sec^2\theta.
\]
The same witness also verifies the constant simplification
\[
  \frac{A^4}{2\operatorname{area}(P_5)^2}
  =
  \frac{5+2\sqrt{5}}{10}.
\]
This proves the claimed secant law for the active $2$-bounce candidate and the
resulting upper bound on the systolic ratio.
\end{proof}
\end{unverified}

\begin{unverified}
\begin{lemma}[Strict-feasibility 3-bounce branches are smooth]
\label{lem:pentagon-rotation-three-bounce-smooth}
Fix a 3-bounce cyclic permutation~$\sigma$ from the billiard enumeration and
define
\[
  U_\sigma
  :=
  \left\{
    \theta \in [0,\pi/10] :
    M_\sigma(\theta) \text{ is non-singular},\;
    \beta_\sigma(\theta) > 0,\;
    Q_\sigma(\theta) > 0
  \right\}.
\]
Then $U_\sigma$ is open in~$[0,\pi/10]$. On each connected component
$I \subset U_\sigma$, the per-orbit action
\[
  A_\sigma(\theta) := \frac{1}{2Q_\sigma(\theta)}
\]
is smooth, hence continuous.
\end{lemma}

\begin{proof}
Let $a(\theta)$ denote the dual-vertex data of~$K_\theta$.
The map $\theta \mapsto a(\theta)$ is smooth because the second factor is
obtained from a fixed pentagon by rotation.
If $\theta_0 \in U_\sigma$, then
$M_\sigma(a(\theta_0))$ is non-singular,
$\beta_\sigma(a(\theta_0)) > 0$,
and $Q_\sigma(a(\theta_0)) > 0$.
Lemma~\ref{lem:orbit-feasibility-open} yields an $\varepsilon > 0$ such that
the same strict inequalities persist for
$|\theta-\theta_0|<\varepsilon$, so $U_\sigma$ is open.
Lemma~\ref{lem:per-orbit-smooth} applied to the smooth parameter
$a(\theta)$ shows that $A_\sigma(\theta)$ is smooth on the same
neighborhood. Since this holds at every point of~$U_\sigma$, the action is
smooth on each connected component~$I \subset U_\sigma$.
\end{proof}
\end{unverified}

\begin{unverified}
\begin{lemma}[Continuity reduction on a strict-feasibility interval]
\label{lem:pentagon-rotation-three-bounce-continuity}
Let~$\sigma$ be a 3-bounce cyclic permutation and let $I \subset U_\sigma$ be
a connected component as in
Lemma~\ref{lem:pentagon-rotation-three-bounce-smooth}. Define
\[
  g(\theta)
  :=
  \left(1+\cos\left(\frac{\pi}{5}\right)\right)^2 \sec(\theta),
\]
the explicit 2-bounce capacity from
Proposition~\ref{prop:pentagon-rotation-two-bounce}. If
\[
  A_\sigma(\theta) > g(\theta)
  \qquad
  \text{for almost every } \theta \in I,
\]
then
\[
  A_\sigma(\theta) \ge g(\theta)
  \qquad
  \text{for every } \theta \in I.
\]
\end{lemma}

\begin{proof}
By Lemma~\ref{lem:pentagon-rotation-three-bounce-smooth},
$A_\sigma$ is continuous on~$I$, and $g$ is continuous on~$[0,\pi/10]$.
If there were $\theta_0 \in I$ with $A_\sigma(\theta_0) < g(\theta_0)$, then
continuity of $A_\sigma-g$ would imply $A_\sigma(\theta) < g(\theta)$ on an
open neighborhood of~$\theta_0$ inside~$I$. That neighborhood has positive
measure, contradicting the assumed almost-everywhere strict inequality.
\end{proof}
\end{unverified}

\begin{unverified}
\begin{lemma}[Reduction of the 3-bounce blocker to branch interiors and endpoints]
\label{lem:pentagon-rotation-three-bounce-reduction}
Assume the following.
\begin{enumerate}
\item For every 3-bounce cyclic permutation~$\sigma$ and every connected
  component $I \subset U_\sigma$, one has
  $A_\sigma(\theta) > g(\theta)$ for almost every $\theta \in I$, where
  $g(\theta)$ is the explicit 2-bounce capacity from
  Lemma~\ref{lem:pentagon-rotation-three-bounce-continuity}.
\item For every admissible 3-bounce orbit at an angle
  $\theta_0 \in [0,\pi/10]$ that does not lie in any such strict-feasibility
  component, either:
  \begin{enumerate}
  \item some $\beta_k(\theta_0)=0$, so repeated use of
    Lemma~\ref{lem:orbit-contraction} contracts the orbit to a shorter
    admissible orbit with the same action, and that contracted orbit is shown
    separately to have action at least~$g(\theta_0)$;
    or
  \item that orbit is shown separately not to minimize.
  \end{enumerate}
\end{enumerate}
Then every minimizing 3-bounce trajectory on $0 \le \theta \le \pi/10$ has
$T^\circ$-length at least that of the minimizing 2-bounce family.
\end{lemma}

\begin{proof}
Fix an admissible 3-bounce orbit at angle~$\theta_0$ and choose a cyclic
permutation~$\sigma$ representing it.
If $\theta_0 \in U_\sigma$, then $\theta_0$ lies in a connected component
$I \subset U_\sigma$, and
Lemma~\ref{lem:pentagon-rotation-three-bounce-continuity} gives
$A_\sigma(\theta_0) \ge g(\theta_0)$.
If $\theta_0 \notin U_\sigma$, then assumption~(2) applies.
In case~(a), Lemma~\ref{lem:orbit-contraction} reduces the action to a
shorter orbit with the same value, and the remaining comparison with
$g(\theta_0)$ is part of assumption~(2a).
In case~(b), the orbit is not minimizing by assumption.
Thus no minimizing 3-bounce orbit can have action below~$g(\theta)$ on the
half-domain.
\end{proof}
\end{unverified}

\begin{unverified}
\begin{lemma}[The first shortlisted 3-bounce branch stays above the 2-bounce candidate]
\label{lem:pentagon-rotation-three-bounce-first-family}
Let
\[
  c := \cos\left(\frac{\pi}{5}\right),
  \qquad
  n_k := \left(
    \cos\left(\frac{\pi}{2} + \frac{2\pi k}{5}\right),
    \sin\left(\frac{\pi}{2} + \frac{2\pi k}{5}\right)
  \right),
  \qquad
  m_k(\theta) := R(\theta)n_k.
\]
Consider the affine 3-bounce support pattern
\[
  q : (e_0,e_4,v_{12}),
  \qquad
  p : (e_2,e_1,w_{40}(\theta)),
\]
where
\[
  e_k := \left\{ q : \langle n_k, q \rangle = c \right\},
  \qquad
  v_{ij} := e_i \cap e_j,
  \qquad
  w_{ij}(\theta) := R(\theta)v_{ij}.
\]
This is a convenient representative of the canonical affine signature
\texttt{Q:0-1-23\ \ P:2-3-01}.
Let
\[
  z := v_{12},
  \qquad
  x := z - a(\theta)m_2(\theta) \in e_0,
  \qquad
  y := x - b(\theta)m_1(\theta) \in e_4,
\]
with $a(\theta), b(\theta) > 0$.
Then the affine product action of this branch is
\[
  A_1(\theta)
  =
  g(\theta)
  +
  \frac{
    5\sin\left(\theta+\frac{\pi}{10}\right)
    \sin\left(\frac{\pi}{10}-\theta\right)
  }
  {
    4\sin^2\left(\theta+\frac{3\pi}{10}\right)\cos\theta
  },
\]
where
\[
  g(\theta)
  :=
  \left(1+\cos\left(\frac{\pi}{5}\right)\right)^2 \sec(\theta).
\]
In particular,
\[
  A_1(\theta) > g(\theta)
  \qquad
  \text{for } 0 \le \theta < \frac{\pi}{10},
\]
and $A_1(\pi/10)=g(\pi/10)$.
\end{lemma}

\begin{proof}
Write
\[
  A := 1+\cos\left(\frac{\pi}{5}\right).
\]
The geometric reduction for this branch is the following.
Since $z=v_{12}=(-\cos(\pi/10),-\sin(\pi/10))$ and $x \in e_0$, one has
\[
  c
  =
  \langle n_0, x \rangle
  =
  \langle n_0, z \rangle - a(\theta)\langle n_0, m_2(\theta) \rangle
  =
  -\sin\left(\frac{\pi}{10}\right)
  +
  a(\theta)\sin\left(\theta+\frac{3\pi}{10}\right).
\]
Using
\[
  \cos\left(\frac{\pi}{5}\right) + \sin\left(\frac{\pi}{10}\right)
  =
  \frac{\sqrt{5}}{2},
\]
this gives
\[
  a(\theta)
  =
  \frac{\sqrt{5}}
       {2\sin\left(\theta+\frac{3\pi}{10}\right)}.
\]
Also, $y \in e_4$ implies
\begin{align*}
  c
  &=
  \langle n_4, y \rangle \\
  &=
  \langle n_4, z \rangle
  -
  a(\theta)\langle n_4, m_2(\theta) \rangle
  -
  b(\theta)\langle n_4, m_1(\theta) \rangle \\
  &=
  -1
  +
  a(\theta)\cos\left(\theta+\frac{\pi}{5}\right)
  +
  b(\theta)\sin\left(\theta+\frac{3\pi}{10}\right),
\end{align*}
because
\[
  \langle n_4, z \rangle = -1,
  \qquad
  \langle n_4, m_2(\theta) \rangle = -\cos\left(\theta+\frac{\pi}{5}\right),
  \qquad
  \langle n_4, m_1(\theta) \rangle = -\sin\left(\theta+\frac{3\pi}{10}\right).
\]
Hence
\[
  b(\theta)\sin\left(\theta+\frac{3\pi}{10}\right)
  =
  A - a(\theta)\cos\left(\theta+\frac{\pi}{5}\right).
\]
\[
  \langle z-y, -w_{40}(\theta) \rangle
  =
  a(\theta) + \sin\left(\frac{\pi}{10}\right)b(\theta),
\]
so the affine product action reduces to
\[
  A_1(\theta)
  =
  c\,a(\theta) + c\,b(\theta) + \langle z-y, -w_{40}(\theta) \rangle
  =
  A\,a(\theta) + \frac{\sqrt{5}}{2}b(\theta).
\]
Here we used
\[
  \cos\left(\frac{\pi}{5}\right) + \sin\left(\frac{\pi}{10}\right)
  =
  \frac{\sqrt{5}}{2}.
\]

At this point the remaining elimination of $b(\theta)$ and the simplification
of $A_1(\theta)-g(\theta)$ are routine symbolic algebra. By the exact CAS
witness from Remark~\ref{rem:pentagon-rotation-cas-witness},
\[
  A_1(\theta) - g(\theta)
  =
  \frac{
    5\sin\left(\theta+\frac{\pi}{10}\right)
    \sin\left(\frac{\pi}{10}-\theta\right)
  }
  {
    4\sin^2\left(\theta+\frac{3\pi}{10}\right)\cos\theta
  }.
\]
On $0 \le \theta < \pi/10$, every denominator factor is positive, and so are
both sine factors. At $\theta=\pi/10$ the second sine factor vanishes. This
proves the claim.
\end{proof}
\end{unverified}

\begin{lemma}[The second shortlisted 3-bounce branch stays above the 2-bounce candidate]
\label{lem:pentagon-rotation-three-bounce-second-family}
Keep the notation of
Lemma~\ref{lem:pentagon-rotation-three-bounce-first-family}. Consider the

\begin{unverified}
