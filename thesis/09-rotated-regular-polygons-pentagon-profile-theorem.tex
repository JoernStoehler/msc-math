\subsection{Exact profile}
\label{subsec:rotated-regular-polygons-formula-for-the-pentagon-product}

For \(\theta\in\mathbb R\), define its distance from the nearest multiple of
\(\pi/5\) by
\[
  d(\theta)=\min_{k\in\mathbb Z}
    \left|\theta-\frac{k\pi}{5}\right|.
\]
Thus \(0\le d(\theta)\le\pi/10\).

\begin{theorem}[Rotated-pentagon profile]
\label{thm:rotated-pentagon-product-formula}
For every \(\theta\in\mathbb R\),
\[
  \operatorname{sys}(K_\theta)
  =
  \frac{5+2\sqrt5}{10}\sec^2\!\bigl(d(\theta)\bigr).
\]
Consequently, the maximum within this family is attained exactly when
\(d(\theta)=\pi/10\), and its value is
\[
  \max_{\theta\in\mathbb R}\operatorname{sys}(K_\theta)
  =\frac{3+\sqrt5}{5}>1.
\]
\end{theorem}

\begin{proof}
\noindent\textbf{Symmetry reduction.}
The profile has period \(2\pi/5\), because
\(R(\theta+2\pi/5)P_5=R(\theta)P_5\).
Simultaneous reflection of both factors gives invariance under
\(\theta\mapsto-\theta\), hence a first reduction to \(0\le\theta\le\pi/5\).

There is a further symmetry because the factors are equal and the pentagon is
odd-sided.  The symplectic map \(-J_0\) sends
\[
  P_5\times_L R(\theta)P_5
  \quad\text{to}\quad
  R(\theta)P_5\times_L(-P_5).
\]
After rotating both factors by \(-\theta\), and using
\(-P_5=R(\pi)P_5=R(\pi/5)P_5\) modulo the pentagon's \(2\pi/5\)-symmetry, we
obtain \(K_{\pi/5-\theta}\).  It is therefore enough to prove
\begin{equation}
\label{eq:rotated-pentagon-half-profile}
  \operatorname{sys}(K_\theta)
  =\frac{5+2\sqrt5}{10}\sec^2\theta,
  \qquad 0\le\theta\le\frac{\pi}{10}.
\end{equation}

\par\medskip\noindent\textbf{The candidate branch.}
We use the facet convention of the executable certificate.  Put
\[
  h=\cos\frac{\pi}{5},\qquad
  n_i=
  \left(
    \cos\left(\frac{\pi}{2}+\frac{2\pi i}{5}\right),
    \sin\left(\frac{\pi}{2}+\frac{2\pi i}{5}\right)
  \right).
\]
In the coordinate order \((q_1,q_2,p_1,p_2)\), the dual vertices of the ten
facets are
\[
  a_i=\frac1h(n_i,0),\qquad
  a_{i+5}=\frac1h(0,R(\theta)n_i),\qquad 0\le i<5.
\]
Consider the alternating two-bounce word with \(q\)-blocks \((3)\), \((1,0)\)
and \(p\)-blocks \((8)\), \((5,6)\); its flattened word is
\[
  \sigma_{\mathrm{act}}=(3,8,1,0,5,6).
\]
In this order, set
\[
  \beta_{\mathrm{act}}
  =\left(
    \frac{\sqrt5}{10},\frac{\sqrt5}{10},
    \frac{5-\sqrt5}{20},\frac{5-\sqrt5}{20},
    \frac{5-\sqrt5}{20},\frac{5-\sqrt5}{20}
  \right).
\]
Direct substitution gives
\[
  \beta_{\mathrm{act}}\in M_{\sigma_{\mathrm{act}}},
  \qquad
  Q_{\sigma_{\mathrm{act}}}(\beta_{\mathrm{act}})
  =\frac{\cos\theta}
  {2\left(1+\cos(\pi/5)\right)^2}.
\]
All weights are positive.  Since the capacity is \(1/(2Q_{\max})\), this
single feasible word gives
\[
  c_{\mathrm{EHZ}}(K_\theta)
  \le c_{\mathrm{act}}(\theta)
  :=\left(1+\cos\frac{\pi}{5}\right)^2\sec\theta.
\]
Rotation preserves the second factor's area, so
\[
  \operatorname{vol}_4(K_\theta)=\operatorname{area}(P_5)^2,
  \qquad
  \operatorname{area}(P_5)=\frac52\sin\frac{2\pi}{5}.
\]
It follows that
\[
  \frac{c_{\mathrm{act}}(\theta)^2}
       {2\operatorname{vol}_4(K_\theta)}
  =\frac{5+2\sqrt5}{10}\sec^2\theta.
\]
This is the required upper bound in \eqref{eq:rotated-pentagon-half-profile}.
