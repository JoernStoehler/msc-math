\subsection{Facet-removal containment bounds}
\label{sec:product-simplification-bounds}

\begin{unverified}
\begin{theorem}[EHZ monotonicity]\label{thm:ehz-monotonicity}
If $K,L\subset\R^4$ are convex bodies and $K\subset L$, then
\[
  c_{\mathrm{EHZ}}(K) \le c_{\mathrm{EHZ}}(L).
\]
\end{theorem}
\end{unverified}

\begin{proof}
This is the monotonicity axiom for symplectic capacities, applied to
$c_{\mathrm{EHZ}}$.  The statement is recorded here so the experiment-facing
distortion bound has a local dependency surface.  A future cleanup should cite
the exact theorem or move this axiom to the shared capacity preliminaries.
\end{proof}

\begin{unverified}
\begin{lemma}[H-representation facet-removal containment]
\label{lem:hrep-facet-removal-containment}
Let $a_1,\ldots,a_F\in\R^4$, and let
\[
  P=\{x\in\R^4:\langle a_i,x\rangle\le 1
      \text{ for }i=1,\ldots,F\}.
\]
Let $R\sqcup S=\{1,\ldots,F\}$ and
\[
  P_R=\{x\in\R^4:\langle a_i,x\rangle\le 1
      \text{ for }i\in R\}
\]
be the polytope obtained by removing the constraints indexed by~$S$.
Assume $\delta\ge0$ satisfies
\[
  \langle a_j,x\rangle\le 1+\delta
  \qquad\text{for every }j\in S\text{ and every }x\in P_R.
\]
Then
\[
  P\subset P_R\subset (1+\delta)P .
\]
\end{lemma}
\end{unverified}

\begin{proof}
The inclusion $P\subset P_R$ is immediate because $P$ satisfies all constraints
and $P_R$ keeps only the constraints in~$R$.

Let $x\in P_R$ and set $\lambda=1+\delta$.  Since $\delta\ge0$, we have
$\lambda\ge1$.  For $i\in R$,
\[
  \langle a_i,x/\lambda\rangle
  =\lambda^{-1}\langle a_i,x\rangle
  \le \lambda^{-1}\le1.
\]
For $j\in S$, the assumed removed-constraint bound gives
\[
  \langle a_j,x/\lambda\rangle
  =\lambda^{-1}\langle a_j,x\rangle
  \le \lambda^{-1}(1+\delta)=1.
\]
Thus $x/\lambda\in P$, hence $x\in\lambda P=(1+\delta)P$.
\end{proof}

\begin{unverified}
\begin{corollary}[Facet-removal capacity, volume, and systolic-ratio bounds]
\label{cor:facet-removal-capacity-volume-sys-bounds}
In the setting of Lemma~\ref{lem:hrep-facet-removal-containment}, assume $P$
and $P_R$ are convex bodies in~$\R^4$, and let $\lambda=1+\delta$.
Then
\[
  1\le
  \frac{c_{\mathrm{EHZ}}(P_R)}{c_{\mathrm{EHZ}}(P)}
  \le \lambda^2
\]
and
\[
  1\le
  \frac{\vol(P_R)}{\vol(P)}
  \le \lambda^4 .
\]
For the systolic ratio
\[
  \sys(K)=\frac{c_{\mathrm{EHZ}}(K)^2}{2\,\vol(K)},
\]
one has
\[
  \lambda^{-4}
  \le
  \frac{\sys(P_R)}{\sys(P)}
  \le
  \lambda^4 .
\]
\end{corollary}
\end{unverified}

\begin{proof}
By Lemma~\ref{lem:hrep-facet-removal-containment},
$P\subset P_R\subset \lambda P$.  Monotonicity
(Theorem~\ref{thm:ehz-monotonicity}) and conformality
(Theorem~\ref{thm:conformality}) give
\[
  c_{\mathrm{EHZ}}(P)
  \le c_{\mathrm{EHZ}}(P_R)
  \le c_{\mathrm{EHZ}}(\lambda P)
  = \lambda^2 c_{\mathrm{EHZ}}(P).
\]
Lebesgue volume is monotone under inclusion and homogeneous of degree~$4$ in
$\R^4$, so
\[
  \vol(P)\le \vol(P_R)\le \vol(\lambda P)=\lambda^4\vol(P).
\]
The lower systolic-ratio bound uses
$c_{\mathrm{EHZ}}(P_R)\ge c_{\mathrm{EHZ}}(P)$ and
$\vol(P_R)\le \lambda^4\vol(P)$:
\[
  \sys(P_R)
  =
  \frac{c_{\mathrm{EHZ}}(P_R)^2}{2\vol(P_R)}
  \ge
  \frac{c_{\mathrm{EHZ}}(P)^2}{2\lambda^4\vol(P)}
  =
  \lambda^{-4}\sys(P).
\]
The upper bound uses
$c_{\mathrm{EHZ}}(P_R)\le\lambda^2c_{\mathrm{EHZ}}(P)$ and
$\vol(P_R)\ge\vol(P)$:
\[
  \sys(P_R)
  \le
  \frac{\lambda^4c_{\mathrm{EHZ}}(P)^2}{2\vol(P)}
  =
  \lambda^4\sys(P).
\]
\end{proof}

\begin{remark}[Contract for the product-simplification experiment]
\label{rem:product-simplification-experiment-contract}
The f64 product-simplification scan reports a number called
\texttt{product\_simplification\_delta\_bound}.  The mathematical meaning
needed by Lemma~\ref{lem:hrep-facet-removal-containment} is:
for every removed original facet $j$ and every point $x$ in the simplified
polytope $P_R$,
\[
  \langle a_j,x\rangle\le
  1+\texttt{product\_simplification\_delta\_bound}.
\]
If that reported number is a valid upper bound and is nonnegative, then the
reported distortion factors
\[
  (1+\delta)^2,\qquad (1+\delta)^4,\qquad
  (1+\delta)^{-4},\qquad (1+\delta)^4
\]
for capacity, volume, and systolic ratio follow from
Corollary~\ref{cor:facet-removal-capacity-volume-sys-bounds}.

This remark deliberately does not certify the floating-point computation of
$\delta$.  Such a certificate would require outward-rounded optimization or an
exact/rational audit of the simplified product factors.
\end{remark}
