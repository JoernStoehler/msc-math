\usepackage{mdframed}

% Approval status bars for math review
% - unverified: red bar — unapproved math, needs Jörn's review
% - notationupdated: orange bar — was approved, mechanical substitution applied
% - approved: no wrapper needed (default appearance)
\newmdenv[
  leftline=true, rightline=false, topline=false, bottomline=false,
  linecolor=red, linewidth=2pt, innerleftmargin=8pt,
  skipabove=0pt, skipbelow=0pt
]{unverified}

\newmdenv[
  leftline=true, rightline=false, topline=false, bottomline=false,
  linecolor=orange, linewidth=2pt, innerleftmargin=8pt,
  skipabove=0pt, skipbelow=0pt
]{notationupdated}

% Neutral sidebars for exact-computation checks used in proof text.
\newmdenv[
  leftline=true, rightline=false, topline=false, bottomline=false,
  linecolor=black!45, linewidth=1.5pt, innerleftmargin=8pt,
  innerrightmargin=0pt, innertopmargin=3pt, innerbottommargin=3pt,
  skipabove=6pt, skipbelow=6pt
]{certificatecheck}
