\begin{lemma}[Dismissal of redundant systems]%
\label{lem:rank-deficiency-dismissal}
Let $(S, \sigma)$ be a pair for which the linear
system~\eqref{eq:linear-system} has a solution
$(\beta_0, \lambda_0, \nu_0)$ with $\beta_0 \geq 0$,
and suppose the null space of the system matrix contains an element
$(\delta\beta, \delta\lambda, \delta\nu)$ with
$\delta\beta \neq 0$.

Then there exist $k \in S$ and a solution
$(\beta^*, \lambda^*, \nu^*)$ with
\begin{enumerate}
\item $\beta^*_k = 0$,
\item $\beta^*_j \geq 0$ for all $j \in S$,
\item $\beta^{*\top} H\, \beta^* = \beta_0^\top H\, \beta_0$.
\end{enumerate}
The restriction $\beta^*|_{S'}$ with $S' = S \setminus \{k\}$
lies in the constraint set $M(K)$ restricted to~$S'$,
so the algorithm for~$(S', \sigma|_{S'})$ maximises
$\beta^\top H'\, \beta$ over at least the value~$\nu_0$.
Consequently, the algorithm may discard any pair $(S, \sigma)$
whose system has such a null-space direction
without missing the capacity-achieving pair.
\end{lemma}

\begin{proof}
If $\beta_0$ already has $\beta_{0,k} = 0$ for some $k \in S$,
take $\beta^* = \beta_0$: conclusions (1)--(3) hold immediately
(with~(3) trivial), and the argument of Step~5 below gives the
restriction to~$S'$. So assume $\beta_{0,i} > 0$ for all~$i$.

Since $(\delta\beta, \delta\lambda, \delta\nu)$ is in the null space:
\[
  \text{(I$'$)}\ N^\top \delta\beta = 0, \qquad
  \text{(II$'$)}\ H\,\delta\beta
    = N\,\delta\lambda + \eta\,\delta\nu, \qquad
  \text{(III$'$)}\ \eta^\top \delta\beta = 0.
\]

\emph{Step~1: $\delta\beta$ has mixed signs.}
Expanding~(III$'$):
$\sum_{i \in S} \eta_i\, \delta\beta_i = 0$
where $\eta_i = h_{\sigma(i)} > 0$ are the heights.
Since $\delta\beta \neq 0$, at least one component is
nonzero, and the vanishing weighted sum with positive weights
forces $\delta\beta$ to have at least one strictly positive
and at least one strictly negative component.

\emph{Step~2: Walk along the null direction.}
Define $\beta(\alpha) = \beta_0 + \alpha\,\delta\beta$.
For every~$\alpha$, the triple
$(\beta(\alpha),\, \lambda_0 + \alpha\,\delta\lambda,\,
  \nu_0 + \alpha\,\delta\nu)$
satisfies~\eqref{eq:linear-system} (linearity of the system).
By Lemma~\ref{lem:well-defined}
(all solutions yield the same value of
$\beta^\top H\, \beta$),
$\beta(\alpha)^\top H\, \beta(\alpha) = \nu_0 + \alpha\,\delta\nu$.
On the other hand, expanding the quadratic form and using
$\delta\beta^\top H\, \beta_0 = 0$ and
$\delta\beta^\top H\, \delta\beta = 0$
(both from the proof of Lemma~\ref{lem:well-defined}):
\[
  \beta(\alpha)^\top H\, \beta(\alpha)
  = \beta_0^\top H\, \beta_0
    + 2\alpha\,\delta\beta^\top H\,\beta_0
    + \alpha^2\,\delta\beta^\top H\,\delta\beta
  = \nu_0.
\]
Comparing the two expressions: $\delta\nu = 0$, so the
multiplier~$\nu$ is the same for every solution.

\emph{Step~3: Identify the boundary point.}
Since $\beta_{0,i} > 0$ for all~$i$,
$\beta(\alpha)$ remains strictly positive for all
$|\alpha|$ sufficiently small.
For each $i$ with $\delta\beta_i < 0$, the component
$\beta_i(\alpha)$ reaches zero at
$\alpha_i = \beta_{0,i} / |\delta\beta_i|$.
Define
\[
  \alpha^*
  = \min\bigl\{\,
      \alpha_i : i \in S,\; \delta\beta_i < 0
    \bigr\},
\]
and let $k$ be an index achieving this minimum.
Such an index exists because $\delta\beta$ has at least one
negative component (Step~1).
Note $\alpha^* > 0$ since $\beta_{0,i} > 0$ and
$|\delta\beta_i| > 0$ for each candidate~$i$.

\emph{Step~4: Verify non-negativity.}
Set $\beta^* = \beta(\alpha^*)$.
Then $\beta^*_k = 0$ by construction.
For $j \neq k$:
\begin{itemize}
\item If $\delta\beta_j \geq 0$:
  $\beta^*_j = \beta_{0,j} + \alpha^*\,\delta\beta_j
    \geq \beta_{0,j} > 0$.
\item If $\delta\beta_j < 0$:
  $\alpha^* \leq \alpha_j = \beta_{0,j}/|\delta\beta_j|$
  (since $\alpha^*$ is the minimum), so
  $\beta^*_j
    = \beta_{0,j} - \alpha^*\,|\delta\beta_j|
    \geq \beta_{0,j} - \alpha_j\,|\delta\beta_j| = 0$.
\end{itemize}
Hence $\beta^*_j \geq 0$ for all $j \in S$.

\emph{Step~5: The smaller pair achieves the same value.}
Since $\beta^*_k = 0$, the quadratic form
$\beta^{*\top} H\, \beta^* = \nu_0$ (Step~2)
receives no contribution from index~$k$.
Let $H'$ denote the action matrix for
$(S', \sigma|_{S'})$, where $S' = S \setminus \{k\}$.
The sub-matrix of~$H$ at rows and columns~$S'$
equals~$H'$ (both are defined by the same normals
$n_{\sigma(j)}$, $j \in S'$), so
$(\beta^*|_{S'})^\top H'\, (\beta^*|_{S'}) = \nu_0$.

The restriction $\beta^*|_{S'}$ satisfies:
$\sum_{j \in S'} \beta^*_j\, n_{\sigma(j)} = 0$
(from $N^\top \beta^* = 0$ and $\beta^*_k = 0$),
$\sum_{j \in S'} \beta^*_j\, h_{\sigma(j)} = 1$
(from $\eta^\top \beta^* = 1$ and $\beta^*_k = 0$),
and $\beta^*_j \geq 0$ for $j \in S'$ (Step~4).
Hence $\beta^*|_{S'} \in M(K)$ restricted to~$S'$.
The algorithm evaluates
$\max\, \beta^\top H'\, \beta$ over this constraint set
and therefore finds value $\geq \nu_0$.
\end{proof}

\begin{remark}[Geometric characterisation of the null space]%
\label{rem:nullspace-geometry}
The null space of the system matrix always contains the subspace
$\{(0, \delta\lambda, 0) : N\,\delta\lambda = 0\}$, which has
dimension $4 - \operatorname{rank}(N)$.
In particular, a null-space element with $\delta\beta = 0$
exists if and only if the visited normals
$\{n_{\sigma(i)} : i \in S\}$ do not span~$\mathbb{R}^4$.
(Proof: if $\delta\beta = 0$, the null-space equations reduce to
$N\,\delta\lambda + \eta\,\delta\nu = 0$; multiplying by any
$\beta_0 \geq 0$ with $\eta^\top\!\beta_0 = 1$ gives $\delta\nu = 0$,
leaving $N\,\delta\lambda = 0$.)
The algorithm does not need this characterisation: it suffices
to inspect the null-space vectors from the SVD and check whether
$\delta\beta \neq 0$.
\end{remark}

\begin{lemma}[Dismissal via constraint-satisfying direction]%
\label{lem:constraint-direction-dismissal}
Let $(S, \sigma)$ be a pair for which the linear
system~\eqref{eq:linear-system} has a solution
$(\beta_0, \lambda_0, \nu_0)$ with $\beta_{0,i} > 0$ for all
$i \in S$.
Let $\delta\beta \in \mathbb{R}^{|S|} \setminus \{0\}$ satisfy
the constraint equations
\[
  N^\top \delta\beta = 0, \qquad \eta^\top \delta\beta = 0.
\]
Define the walk-to-boundary quantities:
for each~$i$ with $\delta\beta_i < 0$, set
$\alpha_i = \beta_{0,i} / |\delta\beta_i|$; then
$\alpha^* = \min\{\alpha_i : \delta\beta_i < 0\}$,
$k$ is an index achieving this minimum,
$\beta^* = \beta_0 + \alpha^*\,\delta\beta$,
and $S' = S \setminus \{k\}$.

Then:
\begin{enumerate}
\item $\delta\beta$ has mixed signs
  (at least one positive and one negative component).
\item $\beta^*_k = 0$, $\beta^*_j \geq 0$ for all $j \in S$,
  and $\beta^*|_{S'}$ is exactly feasible:
  $N^\top \beta^* = 0$ and $\eta^\top \beta^* = 1$
  (which, since $\beta^*_k = 0$, are equivalent to
  the constraints for~$(S', \sigma|_{S'})$).
\item The action value satisfies
  \begin{equation}\label{eq:walk-q-formula}
    \beta^{*\top} H\, \beta^*
    = \nu_0 + (\alpha^*)^2\,
      \delta\beta^\top H\, \delta\beta.
  \end{equation}
\end{enumerate}
In particular, the $(S', \sigma|_{S'})$ algorithm finds value
at least $\nu_0 - (\alpha^*)^2\,
|\delta\beta^\top H\, \delta\beta|$.
\end{lemma}

\begin{proof}
\emph{Part~(1).}
Since $\eta_i = h_{\sigma(i)} > 0$ for all~$i$,
$\sum \eta_i\,\delta\beta_i = 0$ with $\delta\beta \neq 0$
forces at least one positive and one negative component.

\emph{Part~(2).}
The walk-to-boundary construction and non-negativity argument
are identical to Steps~3--4 of
Lemma~\ref{lem:rank-deficiency-dismissal}
(they use only $\beta_0 > 0$, $\delta\beta$ having mixed signs,
and the definition of~$\alpha^*$).
Exact feasibility follows from the constraint equations:
\[
  N^\top\beta^*
  = N^\top\beta_0 + \alpha^*\,N^\top\delta\beta
  = 0 + 0 = 0, \qquad
  \eta^\top\beta^*
  = \eta^\top\beta_0 + \alpha^*\,\eta^\top\delta\beta
  = 1 + 0 = 1.
\]
Since $\beta^*_k = 0$, these equalities pass to the
restriction~$\beta^*|_{S'}$.

\emph{Part~(3).}
Expanding the quadratic form:
\[
  \beta^{*\top} H\,\beta^*
  = \beta_0^\top H\,\beta_0
    + 2\alpha^*\,\delta\beta^\top H\,\beta_0
    + (\alpha^*)^2\,\delta\beta^\top H\,\delta\beta.
\]
The first term equals~$\nu_0$.
For the cross term, the KKT condition
$H\,\beta_0 = N\,\lambda_0 + \eta\,\nu_0$ gives
\[
  \delta\beta^\top H\,\beta_0
  = (N^\top\delta\beta)^\top\lambda_0
    + (\eta^\top\delta\beta)\,\nu_0
  = 0 + 0 = 0. \qedhere
\]
\end{proof}

\begin{lemma}[Constraint projection of approximate null directions]%
\label{lem:constraint-projection}
Let $(\delta\beta, \delta\lambda, \delta\nu)$ be a unit vector
($\|(\delta\beta, \delta\lambda, \delta\nu)\|_2 = 1$) with
residual
\[
  \begin{pmatrix}
    N^\top & 0 & 0 \\
    H & -N & -\eta \\
    \eta^\top & 0 & 0
  \end{pmatrix}
  \begin{pmatrix}\delta\beta\\\delta\lambda\\\delta\nu\end{pmatrix}
  = \begin{pmatrix}r_\beta\\r_\lambda\\r_\nu\end{pmatrix},
  \qquad
  \varepsilon
  := \bigl\|(r_\beta, r_\lambda, r_\nu)\bigr\|_2.
\]
Let $C = [N \mid \eta] \in \mathbb{R}^{|S| \times 5}$
and suppose $C$ has full column rank with smallest singular
value $\sigma_C > 0$.
Define the corrected direction
\[
  \delta\beta'
  = \delta\beta
    - C\,(C^\top C)^{-1}
      \begin{pmatrix}r_\beta \\ r_\nu\end{pmatrix}.
\]
Then:
\begin{enumerate}
\item $N^\top\delta\beta' = 0$ and
  $\eta^\top\delta\beta' = 0$.
\item $\|\delta\beta - \delta\beta'\|_2
  \leq \varepsilon / \sigma_C$.
\item $\delta\beta' \neq 0$ whenever
  $\|\delta\beta\|_2 > \varepsilon / \sigma_C$.
\end{enumerate}
\end{lemma}

\begin{proof}
Write $\Delta = C\,(C^\top C)^{-1}(r_\beta, r_\nu)^\top$.
Since $N^\top\delta\beta = r_\beta$ and
$\eta^\top\delta\beta = r_\nu$, and
$C^\top\Delta = (r_\beta, r_\nu)^\top$ by construction:
\[
  C^\top\delta\beta'
  = C^\top\delta\beta - (r_\beta, r_\nu)^\top
  = (r_\beta, r_\nu)^\top - (r_\beta, r_\nu)^\top = 0,
\]
giving~(1).
For~(2):
$\|\Delta\|
= \|C\,(C^\top C)^{-1}(r_\beta, r_\nu)^\top\|
\leq \sigma_C^{-1}\,\|(r_\beta, r_\nu)\|
\leq \varepsilon / \sigma_C$.
Part~(3) follows by the triangle inequality.
\end{proof}

\begin{proposition}[Approximate dismissal of near-singular systems]%
\label{prop:approximate-dismissal}
Let $(S, \sigma)$ be a pair for which the linear
system~\eqref{eq:linear-system} has a solution
$(\beta_0, \lambda_0, \nu_0)$ with $\beta_{0,i} > 0$ for all
$i \in S$.
Let $(\delta\beta, \delta\lambda, \delta\nu)$ be a unit vector
with residual~$\varepsilon$ as in
Lemma~\ref{lem:constraint-projection},
and suppose $\|\delta\beta\|_2 > \varepsilon / \sigma_C$
(so that the corrected direction~$\delta\beta'$ from
Lemma~\ref{lem:constraint-projection} is non-zero).

Apply Lemma~\ref{lem:constraint-direction-dismissal}
with~$\delta\beta'$: the walk-to-boundary gives
$\beta^*|_{S'} \in M(K)|_{S'}$ with $|S'| = |S| - 1$,
and the $(S', \sigma|_{S'})$ algorithm finds value at least
\[
  \nu_0
    - (\alpha^*)^2\,\|\delta\beta'\|\,
      \Bigl(1 + \frac{\|H\|}{\sigma_C}\Bigr)\,\varepsilon,
\]
where $\|\cdot\|$ denotes the spectral norm.
\end{proposition}

\begin{proof}
Lemma~\ref{lem:constraint-projection} gives $\delta\beta'$
with $N^\top\delta\beta' = 0$,
$\eta^\top\delta\beta' = 0$, and $\delta\beta' \neq 0$.
Lemma~\ref{lem:constraint-direction-dismissal} then gives a
feasible boundary point $\beta^*|_{S'}$ with
\[
  \beta^{*\top} H\,\beta^*
  = \nu_0
    + (\alpha^*)^2\,\delta\beta'^\top H\,\delta\beta'.
\]

It remains to bound
$|\delta\beta'^\top H\,\delta\beta'|$.

Write $\Delta = \delta\beta - \delta\beta'$
with $\|\Delta\| \leq \varepsilon/\sigma_C$
(Lemma~\ref{lem:constraint-projection}).
The second block row of the residual equation gives
$H\,\delta\beta = N\,\delta\lambda + \eta\,\delta\nu + r_\lambda$,
so
\[
  H\,\delta\beta'
  = H\,\delta\beta - H\,\Delta
  = N\,\delta\lambda + \eta\,\delta\nu + r_\lambda - H\,\Delta.
\]
Since $N^\top\delta\beta' = 0$ and $\eta^\top\delta\beta' = 0$:
\begin{align*}
  \delta\beta'^\top H\,\delta\beta'
  &= \underbrace{(N^\top\delta\beta')^\top}_{=\,0}\delta\lambda
    + \underbrace{(\eta^\top\delta\beta')}_{=\,0}\,\delta\nu
    + \delta\beta'^\top(r_\lambda - H\,\Delta) \\
  &= \delta\beta'^\top(r_\lambda - H\,\Delta).
\end{align*}
Therefore
\[
  |\delta\beta'^\top H\,\delta\beta'|
  \leq \|\delta\beta'\|\,
       \bigl(\|r_\lambda\| + \|H\|\,\|\Delta\|\bigr)
  \leq \|\delta\beta'\|\,
       \Bigl(1 + \frac{\|H\|}{\sigma_C}\Bigr)\,\varepsilon,
\]
using $\|r_\lambda\| \leq \varepsilon$ and
$\|\Delta\| \leq \varepsilon/\sigma_C$.

\emph{Conclusion.}
Since the $(S', \sigma|_{S'})$ algorithm maximises over
all of $M(K)|_{S'}$, it finds value at least
\[
  \beta^{*\top} H\,\beta^*
  \geq \nu_0
    - (\alpha^*)^2\,\|\delta\beta'\|\,
      \Bigl(1 + \frac{\|H\|}{\sigma_C}\Bigr)\,\varepsilon.
\]
\end{proof}

\begin{remark}[Algorithmic application]%
\label{rem:approximate-dismissal-algorithm}
In the SVD-based solver,
$(\delta\beta, \delta\lambda, \delta\nu)$ is the right singular
vector for the smallest singular value~$\sigma_{\min}$,
and $\varepsilon = \sigma_{\min}$.
The condition $\|\delta\beta\| > \varepsilon/\sigma_C$ says
that the near-singular direction has a non-trivial
$\beta$-component after constraint projection.
If it fails, the direction is effectively confined to the
multiplier block, $\beta$ is approximately unique, and no
dismissal is needed.

The bound is proportional to
$(\alpha^*)^2\varepsilon$
rather than $\alpha^*\varepsilon$:
the constraint projection eliminates the linear cross-term
$\delta\beta^\top H\,\beta_0$ that would appear without it
(cf.~the proof of
Lemma~\ref{lem:constraint-direction-dismissal},
Part~(3)).
The remaining error can be evaluated numerically for each
near-singular pair.
\end{remark}
